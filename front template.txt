<script>
// Stop requests belonging to the previous answer before showing another question.
if (window.__avgCardSession && window.__avgCardSession.dispose) window.__avgCardSession.dispose();
function autoSaveSpellingInput() {
    const input = document.getElementById('spellCheckInput');
    if (!input) return;
    const text = input.value.trim();
    window.__avgSpellingInput = text;
    try {
        if (text) localStorage.setItem('userSpellingInput', text);
        else localStorage.removeItem('userSpellingInput');
    } catch (e) { /* the back template can still read session memory */ }
}
// Tự động lưu khi người dùng rời khỏi ô input
document.getElementById('spellCheckInput').addEventListener('blur', function() {
    autoSaveSpellingInput();
});

// Tự động lưu khi người dùng nhấn Enter
document.getElementById('spellCheckInput').addEventListener('keypress', function(e) {
    if (e.key === 'Enter') {
        autoSaveSpellingInput();
        // Tự động kích hoạt nút Show Answer nếu có
        if (typeof py !== 'undefined' && py.hasOwnProperty('cmd')) {
            py.cmd('ans');
        }
    }
});
</script>
