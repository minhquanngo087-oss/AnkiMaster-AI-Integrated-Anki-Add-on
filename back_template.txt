function getEdgeVoice(text) {
    const lang = detectLanguage(text);
    switch (lang) {
        case 'zh': return 'zh-CN-XiaoxiaoNeural';
        case 'ja': return 'ja-JP-NanamiNeural';
        case 'ko': return 'ko-KR-SunHiNeural';

async function speakWithEdgeReal(text, event) {
    if (event) event.stopPropagation();
    
    const cleanText = text ? text.replace(/[\(\[\{\<][^\)\]\}\>]*[\)\]\}\>]/g, '').trim() : '';
    if (!cleanText) return;

    let btn = event?.target;
    if (btn && btn.tagName !== 'BUTTON') btn = btn.closest('button');
    
    // --- Cập nhật UI ---
    if (btn) {
        document.querySelectorAll('.tts-btn').forEach(b => {
            b.classList.remove('playing');
            if (b.classList.contains('gemini-btn')) b.innerHTML = '🔊';
            if (b.classList.contains('edge-btn')) b.innerHTML = '⚡';
        });
        
        btn.classList.add('playing');
        btn.innerHTML = '⏹️'; // Icon Stop khi đang đọc (theo code mẫu)
    }

    // --- Dừng các luồng âm thanh cũ ---
    if (window.speechSynthesis) window.speechSynthesis.cancel();
    if (typeof audioCtx !== 'undefined' && audioCtx && audioCtx.state === 'running') {
        try { audioCtx.close(); } catch(e){}
    }

    if (!window.speechSynthesis) return;

    // --- Web Speech API Logic (Thay thế Fetch Edge API để chạy được trên Mobile) ---
    const u = new SpeechSynthesisUtterance(cleanText);
    
    // Phát hiện ngôn ngữ (dựa theo logic code mẫu)
    const isChinese = /[\u4e00-\u9fff\u3400-\u4dbf]/.test(cleanText);
    const lang = isChinese ? 'zh' : 'en';

    // Cấu hình giọng đọc
    if(lang === 'zh'){
        u.lang = 'zh-CN';
        u.rate = 0.8;
        u.pitch = 1.0;
    } else {
        u.lang = 'en-GB'; // Force UK accent
        u.rate = 0.9;
        u.pitch = 1.0;
    }

    // Hàm phát âm thanh
    const play = () => {
        const voices = window.speechSynthesis.getVoices();
        let v = null;
        if(lang === 'zh') {
            v = voices.find(x => x.lang.includes('zh') || x.lang.includes('cn') || x.name.toLowerCase().includes('chinese'));
        } else {
            // Ưu tiên giọng UK (en-GB), fallback sang bất kỳ giọng en nào
            v = voices.find(x => x.lang === 'en-GB')
              || voices.find(x => x.lang.startsWith('en-GB'))
              || voices.find(x => x.name.toLowerCase().includes('british'))
              || voices.find(x => x.lang.startsWith('en'));
        }
        
        if(v) u.voice = v;
        window.speechSynthesis.speak(u);
    };

    // Sự kiện khi đọc xong hoặc lỗi
    const reset = () => {
        if (btn) {
            btn.classList.remove('playing');
            btn.innerHTML = '⚡'; // Trả về icon tia sét ban đầu
        }
    };
    u.onend = reset;
    u.onerror = reset;

    // Xử lý trường hợp danh sách giọng đọc chưa tải xong
    if (window.speechSynthesis.getVoices().length === 0) {
        window.speechSynthesis.onvoiceschanged = play;
    } else {
        play();
    }
}

    if (typeof list === 'string') list = list.split(/[\n,]+/);
    if (!Array.isArray(list)) return [];
    return list.map(function(k) { return String(k).trim(); })
        .filter(function(k, i, keys) { return k.length > 10 && k.indexOf('*') < 0 && keys.indexOf(k) === i; });
}
// kind: 'text' (Luyện câu, Áp dụng) hoặc 'tts'. Khung nào trống thì dùng khung còn lại.
function avgCardKeys(kind) {

    var main = avgCleanKeys(cfg[kind]);
    return main.length ? main : avgCleanKeys(cfg[kind === 'tts' ? 'text' : 'tts']);
}

function stopTTS() {

        audioCtx.close();
        audioCtx = null;
    }
    if (window.speechSynthesis) {
        window.speechSynthesis.cancel();

// Hàm dọn dẹp các request cũ đang treo
function cleanupOldRequests() {
    if (globalTimeoutId) {
        clearTimeout(globalTimeoutId);
        globalTimeoutId = null;
    }
    if (globalTTSAbortController) {
        globalTTSAbortController.abort(); // Ngắt kết nối mạng ngay lập tức
        globalTTSAbortController = null;
    }
    // Dừng Web Speech cũ nếu đang nói
    if (window.speechSynthesis) window.speechSynthesis.cancel();
}

async function speakWithGemini(text, event, keyIndex = 0) {
    if (!text || !text.trim()) return;

    // ========================================================================
    // [NEW] LOGIC ÉP BUỘC ĐỌC THEO IPA CHÍNH XÁC
    // Kiểm tra xem đoạn text đang cần đọc có phải là từ vựng chính không.
    // Nếu đúng, chèn IPA vào prompt để ép Gemini-2.5-pro-audio đọc chuẩn xác.
    // ========================================================================
    let promptForAudio = text;
    const mainWord = document.querySelector('.word')?.textContent?.trim() || '';
    
    if (text.trim() === mainWord) {
        const currentIpa = document.querySelector('.ipa')?.textContent?.trim() || '';
        if (currentIpa && !currentIpa.includes('{' + '{')) {
            // Director-style instruction for TTS: ép giọng UK với IPA
            promptForAudio = `Say the word '${text}' in a British RP accent, matching this pronunciation: /${currentIpa}/.`;
        } else {
            promptForAudio = `Say the word '${text}' in a British RP accent.`;
        }
    } else {
        const isEnglish = detectLanguage(text) === 'en';
        if (isEnglish) {
            promptForAudio = `Read this in a British RP accent: ${text}`;
        }
    }
    // ========================================================================
    
    // Lần thử đầu lấy danh sách key mới; các lần thử key tiếp theo dùng lại đúng danh sách đó
    if (keyIndex === 0 || !ttsKeySnapshot) ttsKeySnapshot = getGeminiTTSKeys();
    var keys = ttsKeySnapshot;

    // Nếu không có key, fallback ngay
    if (keys.length === 0) {
        speakWithWebSpeechFallback(text, event);
        return;
    }

    var currentKey = keys[keyIndex % keys.length];

    // --- XỬ LÝ GIAO DIỆN (Chỉ lần đầu) ---
    if (keyIndex === 0) {
        if (audioCtx && audioCtx.state === 'running') {
            try { audioCtx.close(); } catch(e){}
            audioCtx = null;
        }
        
        document.querySelectorAll('.tts-btn, .tts-btn-small').forEach(function(b) {
            b.classList.remove('playing');
            if (b.classList.contains('gemini-btn')) b.innerHTML = '🔊';
            if (b.classList.contains('edge-btn')) b.innerHTML = '⚡';
            if (!b.classList.contains('gemini-btn') && !b.classList.contains('edge-btn')) b.innerHTML = '🔊';
        });
        
        var button = event?.target || document.querySelector('.tts-btn, .tts-btn-small');
        if (button && button.tagName !== 'BUTTON') button = button.closest('button');
        currentGeminiTTSButton = button;
        
        if (button) {
            button.classList.add('playing');
            button.innerHTML = '⏳';
        }
    } else {
        var button = currentGeminiTTSButton;
    }
    
    // --- CƠ CHẾ TIMEOUT CHẮC CHẮN CHO MOBILE ---
    let isRequestFinished = false; // Cờ kiểm soát: true = đã xong (thành công hoặc lỗi)
    const controller = new AbortController();
    
    // Thiết lập bộ đếm giờ 5 giây
    const timeoutId = setTimeout(() => {
        if (isRequestFinished) return; // Nếu API đã trả về thì thôi
        
        console.warn("⚠️ Hết giờ (5s)! Bắt đầu chuyển sang giọng đọc máy...");
        isRequestFinished = true; // Đánh dấu là xong để chặn API nếu nó trả về sau đó
        controller.abort();       // Ngắt mạng
        
        // Chuyển ngay sang Fallback
        resetTTSIcons(); 
        speakWithWebSpeechFallback(text, event);
    }, 5000); 

    try {
        console.log(`Đang gọi Gemini TTS (Key #${keyIndex + 1})...`);
        var url = `https://generativelanguage.googleapis.com/v1beta/models/${avgTtsModel()}:generateContent?key=${currentKey}`;
        
        // Xác định languageCode dựa trên ngôn ngữ: en-GB cho tiếng Anh, 
        // các ngôn ngữ khác dùng locale tương ứng
        var detectedLang = detectLanguage(text);
        var ttsLangCode = 'en-GB'; // Mặc định UK English
        if (detectedLang === 'zh') ttsLangCode = 'zh-CN';
        else if (detectedLang === 'ja') ttsLangCode = 'ja-JP';
        else if (detectedLang === 'vi') ttsLangCode = 'vi-VN';
        else if (detectedLang === 'ko') ttsLangCode = 'ko-KR';
        
        var requestBody = {
            contents: [{parts: [{text: promptForAudio}]}],
            generationConfig: {
                responseModalities: ["AUDIO"],
                speechConfig: {
                    voiceConfig: { prebuiltVoiceConfig: { voiceName: GEMINI_TTS_VOICE } },
                    languageCode: ttsLangCode
                }
            }
        };
        
        console.log(`🇬🇧 TTS Config: lang=${ttsLangCode}, voice=${GEMINI_TTS_VOICE}`);
        
        var response = await fetch(url, {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify(requestBody),
            signal: controller.signal
        });

        // Nếu code chạy đến đây tức là chưa bị timeout ngắt
        if (isRequestFinished) return; // Nếu timeout đã chạy trước đó thì bỏ qua kết quả này
        clearTimeout(timeoutId);       // Hủy bộ đếm giờ vì đã có kết quả

        if (!response.ok) {
            // Key bị xoá/thu hồi -> ghi nhớ để lần sau bỏ qua (dùng chung với Luyện câu, Áp dụng)
            if (response.status === 400) {
                const err = await geminiReadError(response);
                if (/API_KEY/.test(err.reason + err.message)) geminiMark('dead', geminiKeyId(currentKey), 24 * 3600 * 1000);
            }
            // Nếu lỗi server/limit và còn key -> Thử key tiếp theo
            if ([429, 500, 503, 403].includes(response.status)) {
                if (keyIndex < keys.length - 1) {
                    await speakWithGemini(text, event, keyIndex + 1);
                    return;
                }
            }
            throw new Error(`Lỗi API: ${response.status}`);
        }

        var data = await response.json();
        
        if (data.candidates && data.candidates[0].content && data.candidates[0].content.parts[0].inlineData) {
            var audioBase64 = data.candidates[0].content.parts[0].inlineData.data;
            
            // Đánh dấu đã xong để timeout không chạy nữa
            isRequestFinished = true; 
            await playPcmAudio(audioBase64, button, text);
        } else {
            throw new Error("Không có audio data");
        }

    } catch (e) {
        if (isRequestFinished) return; // Nếu timeout đã xử lý rồi thì thôi
        
        isRequestFinished = true;
        clearTimeout(timeoutId);
        
        console.error("Gemini TTS Error:", e);
        
        // Nếu lỗi mạng (không phải do timeout ngắt) và còn key -> Thử tiếp
        if (e.name !== 'AbortError' && keyIndex < keys.length - 1) {
             await speakWithGemini(text, event, keyIndex + 1);
             return;
        }
        
        // Fallback cuối cùng
        resetTTSIcons();
        speakWithWebSpeechFallback(text, event);
    }
}

// Hàm giải mã và phát âm thanh PCM
async function playPcmAudio(base64String, button, originalText) {
    try {
        var binaryString = atob(base64String);
        var len = binaryString.length;
        var bytes = new Uint8Array(len);
        for (var i = 0; i < len; i++) {
            bytes[i] = binaryString.charCodeAt(i);
        }
        
        var int16Data = new Int16Array(bytes.buffer);
        var float32Data = new Float32Array(int16Data.length);
        for (var i = 0; i < int16Data.length; i++) {
            float32Data[i] = int16Data[i] / 32768.0;
        }

        if (!audioCtx || audioCtx.state === 'closed') {
            audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
        
        // QUAN TRỌNG: Resume context cho Mobile
        if (audioCtx.state === 'suspended') {
            await audioCtx.resume();
        }
        
        var buffer = audioCtx.createBuffer(1, float32Data.length, 24000);
        buffer.copyToChannel(float32Data, 0);
        
        var source = audioCtx.createBufferSource();
        source.buffer = buffer;
        source.connect(audioCtx.destination);
        
        source.onended = function() {
            resetTTSIcons();
        };
        
        if (button) {
            if (button.classList.contains('gemini-btn')) {
                button.innerHTML = '🔊';
            } else if (button.classList.contains('edge-btn')) {
                button.innerHTML = '⚡';
            } else {
                button.innerHTML = '🔊';
            }
        }
        
        source.start(0);
        
    } catch (e) {
        console.error("Audio Playback Error:", e);
        resetTTSIcons();
        speakWithWebSpeechFallback(originalText, null);
    }
}

function resetTTSIcons() {
    document.querySelectorAll('.tts-btn, .tts-btn-small').forEach(function(b) {
        b.classList.remove('playing');
        if (b.classList.contains('gemini-btn')) b.innerHTML = '🔊';
        else if (b.classList.contains('edge-btn')) b.innerHTML = '⚡';
        else b.innerHTML = '🔊';
    });
    currentGeminiTTSButton = null;
}

// Fallback: Web Speech API (Cực kỳ đơn giản và nhanh)
function speakWithWebSpeechFallback(text, event) {
    if (!window.speechSynthesis) return;
    
    console.log(">> Running Fallback TTS");
    
    // Đảm bảo dừng mọi thứ trước đó
    window.speechSynthesis.cancel();
    
    const u = new SpeechSynthesisUtterance(text);
    
    // Cố gắng chọn giọng tốt nhất
    const lang = detectLanguage(text);
    if (lang === 'zh') u.lang = 'zh-CN';
    else if (lang === 'vi') u.lang = 'vi-VN';
    else if (lang === 'ja') u.lang = 'ja-JP';
    else {
        u.lang = 'en-GB'; // Force UK accent
        // Tìm giọng UK trong danh sách
        const voices = window.speechSynthesis.getVoices();
        const ukVoice = voices.find(x => x.lang === 'en-GB')
                     || voices.find(x => x.lang.startsWith('en-GB'))
                     || voices.find(x => x.name.toLowerCase().includes('british'));
        if (ukVoice) u.voice = ukVoice;
    }
    
    u.rate = 0.9;
    
    // Hiệu ứng nút
    var btn = currentGeminiTTSButton;
    // Nếu không tìm thấy biến toàn cục, tìm lại qua event
    if (!btn && event && event.target) {
        btn = event.target.closest('button') || event.target;
    }

    if (btn) {
        btn.classList.add('playing');
        btn.innerHTML = '🗣️'; // Icon khác để biết là fallback
        
        u.onstart = function() { 
            if(btn) btn.innerHTML = '⏹️'; 
        };
        u.onend = function() { 
            if(btn) {
                btn.classList.remove('playing');
                if (btn.classList.contains('gemini-btn')) btn.innerHTML = '🔊';
                else if (btn.classList.contains('edge-btn')) btn.innerHTML = '⚡';
                else btn.innerHTML = '🔊';
            }
        };
        u.onerror = function() {
            if(btn) {
                btn.classList.remove('playing');
                if (btn.classList.contains('gemini-btn')) btn.innerHTML = '🔊';
                else if (btn.classList.contains('edge-btn')) btn.innerHTML = '⚡';
                else btn.innerHTML = '🔊';
            }
        };
    }
    
    window.speechSynthesis.speak(u);
}

function detectLanguage(text) {
    if (!text) return 'en';
    if (/[\u4E00-\u9FFF]/.test(text)) return 'zh';
    if (/[àáạảãâầấậẩẫăằắặẳẵèéẹẻẽêềếệểễìíịỉĩòóọỏõôồốộổỗơờớợởỡùúụủũưừứựửữỳýỵỷỹđ]/.test(text.toLowerCase())) return 'vi';
    if (/[\u3040-\u309f\u30a0-\u30ff]/.test(text)) return 'ja';
    return 'en';
}

// Hàm xử lý đọc văn bản tiếng Trung (CẬP NHẬT)
function speakChineseText(event, text) {
    if (event) event.stopPropagation();
    
    // THÊM DÒNG NÀY: Ngắt các luồng âm thanh cũ để tránh lỗi trên Mobile/chồng âm
    if (typeof cleanupOldRequests === 'function') cleanupOldRequests();
    
    var cleanText = text;
    if (text) {
        // Xóa nội dung trong mọi loại ngoặc
        cleanText = cleanText.replace(/[\(\[\{\<][^\)\]\}\>]*[\)\]\}\>]/g, '').trim();
    }
    
    if (!cleanText) cleanText = text;
    
    speakWithGemini(cleanText, event);
}

// Hàm phát hiện ngôn ngữ context - CHỈ 4 NGÔN NGỮ
function detectContextLanguage(text) {
    if (!text) return "Unknown";
    
    // Japanese
    if (/[\u3040-\u30ff\u31f0-\u31ff]/.test(text)) return "Japanese";
    
    // Chinese
    if (/[\u4e00-\u9fff]/.test(text)) return "Chinese";
    
    // Vietnamese (các ký tự có dấu đặc trưng)
    if (/[àáạảãâầấậẩẫăằắặẳẵèéẹẻẽêềếệểễìíịỉĩòóọỏõôồốộổỗơờớợởỡùúụủũưừứựửữỳýỵỷỹđ]/i.test(text)) 
        return "Vietnamese";
    
    // Mặc định là English
    return "English";
}

// Hàm phân loại context trong JavaScript
function categorizeContext(text) {
    const textLower = text.toLowerCase();
    const categories = {
        'medical': ['y học', 'bệnh viện', 'bác sĩ', 'sức khỏe', 'y tế'],
        'technology': ['công nghệ', 'máy tính', 'phần mềm', 'lập trình'],
        'business': ['kinh doanh', 'thương mại', 'công ty', 'doanh nghiệp'],
        'academic': ['học thuật', 'trường học', 'giáo dục', 'nghiên cứu'],
        'daily': ['hàng ngày', 'giao tiếp', 'cuộc sống']
    };
    
    for (const [category, keywords] of Object.entries(categories)) {
        if (keywords.some(keyword => textLower.includes(keyword))) {
            return category;
        }
    }
    return 'general';
}

function getContextCategoryName(category) {
    const names = {
        'medical': 'Y học',
        'technology': 'Công nghệ',
        'business': 'Kinh doanh',
        'academic': 'Học thuật', 
        'daily': 'Hàng ngày',
        'general': 'Chung'
    };
    return names[category] || category;
}

                            ${example.translation ? `<div class="etymology-example-translation">${example.translation}</div>` : ''}

        console.error('Lỗi phân tích dữ liệu etymology examples:', error);
        examplesContainer.style.display = 'none';

            if (html) {
                listEl.innerHTML = html;
                containerEl.style.display = 'block';
                
                // Hiển thị số lượng nghĩa chính xác
                const actualCount = otherMeanings.filter(m => m && m.meaning && m.meaning.trim() !== "").length;
                const titleEl = containerEl.querySelector('.group-title');
                if (titleEl) {
                    if (actualCount === 10) {
                        titleEl.innerHTML = `🔤 10 Nghĩa chính của từ`;
                    } else {
                        titleEl.innerHTML = `🔤 ${actualCount} Nghĩa chính của từ`;
                    }
                }
            } else {
                containerEl.style.display = 'none';
            }
        } else {
            containerEl.style.display = 'none';
        }
    } catch (error) {
        console.error('Lỗi phân tích dữ liệu other meanings:', error);
        containerEl.style.display = 'none';
    }
}

var GEMINI_TEXT_MODELS = [
    { model: 'gemini-3.5-flash-lite', thinking: { thinkingLevel: 'low' } },

var GEMINI_TEXT_TIMEOUT_MS = 25000;
var GEMINI_TOTAL_TIMEOUT_MS = 65000;
var GEMINI_MAX_CALLS = 8;
var GEMINI_HEALTH_KEY = 'gemini_health_v2';
var geminiKeyStart = 0;

function geminiHealth() {
    const h = spStoreGet(GEMINI_HEALTH_KEY) || {};
    return { dead: h.dead || {}, block: h.block || {}, busy: h.busy || {} };
}
function geminiKeyId(key) { return key.slice(-8); }
function geminiMark(kind, id, ms) {
    const h = geminiHealth(), now = Date.now();
    h[kind][id] = now + ms;
    ['dead', 'block', 'busy'].forEach(function(k) {
        Object.keys(h[k]).forEach(function(x) { if (h[k][x] < now) delete h[k][x]; });
    });
    spStoreSet(GEMINI_HEALTH_KEY, h);
}
function geminiIsDeadKey(key) { return (geminiHealth().dead[geminiKeyId(key)] || 0) > Date.now(); }
function geminiError(name, message) {
    const err = new Error(message);
    err.name = name;
    return err;
}
function geminiEnsureActive(session, options) {
    if (!avgSessionIsCurrent(session) || options.cancelled || (options.isCurrent && !options.isCurrent())) {
        throw geminiError('CancelledError', 'Đã dừng yêu cầu.');
    }
}
// One cancellable operation owns its timer, including response-body reads and retry waits.
function geminiOperation(session, options, timeout, start) {
    return new Promise(function(resolve, reject) {
        let finished = false, stop = null;
        const finish = function(err, value) {
            if (finished) return;
            finished = true;
            clearTimeout(timer);
            const i = session.pending.indexOf(cancel);
            if (i >= 0) session.pending.splice(i, 1);
            if (options.cancel === cancel) options.cancel = null;
            if (err && stop) stop();
            if (err) reject(err); else resolve(value);
        };
        const cancel = function() { finish(geminiError('CancelledError', 'Đã dừng yêu cầu.')); };
        const timer = setTimeout(function() { finish(geminiError('TimeoutError', 'Kết nối quá thời gian chờ.')); }, timeout);
        session.pending.push(cancel);
        options.cancel = cancel;
        try {
            geminiEnsureActive(session, options);
            stop = start(function(value) { finish(null, value); }, function(err) { finish(err); });
        } catch (e) { finish(e); }
    });
}
function geminiWait(ms, session, options) {
    return geminiOperation(session, options, ms + 1000, function(resolve) {
        const timer = setTimeout(resolve, ms);
        return function() { clearTimeout(timer); };
    });
}
function geminiRequest(url, body, timeout, session, options) {
    return geminiOperation(session, options, timeout, function(resolve, reject) {
        const controller = typeof AbortController === 'function' ? new AbortController() : null;
        const init = { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: body };
        if (controller) init.signal = controller.signal;
        Promise.resolve().then(function() { return fetch(url, init); }).then(async function(res) {
            const data = await res.json().catch(function() { return null; });
            resolve({ response: res, data: data });
        }).catch(reject);
        return function() { if (controller) controller.abort(); };
    });
}
async function geminiReadError(res, data) {
    // TTS still calls this helper with a Response only.
    if (arguments.length < 2) data = await res.json().catch(function() { return null; });
    const e = (data && data.error) || {}, details = e.details || [];
    const reasons = [], quota = [];
    let retryMs = 0;
    details.forEach(function(d) {
        if (d.reason) reasons.push(d.reason);
        (d.violations || []).forEach(function(v) { quota.push(v.quotaId || ''); });
        if (d.retryDelay) retryMs = Math.max(retryMs, parseFloat(d.retryDelay) * 1000 || 0);
    });
    const retryAfter = res.headers && res.headers.get('Retry-After');
    if (retryAfter) {
        const seconds = Number(retryAfter);
        retryMs = Math.max(retryMs, Number.isFinite(seconds) ? seconds * 1000 : Date.parse(retryAfter) - Date.now() || 0);
    }
    return { status: res.status, reason: reasons[0] || e.status || '', quota: quota.join(' '), message: e.message || '', retryMs: retryMs };
}
function geminiValidateJson(data, schema) {
    if (!schema) return;
    const type = String(schema.type || '').toUpperCase();
    const valid = type === 'OBJECT' ? data && typeof data === 'object' && !Array.isArray(data)
        : type === 'ARRAY' ? Array.isArray(data) : type === 'BOOLEAN' ? typeof data === 'boolean'
        : type === 'STRING' ? typeof data === 'string' : true;
    if (!valid) throw geminiError('InvalidResponseError', 'AI trả về dữ liệu chưa đúng định dạng.');
    if (type === 'OBJECT') {
        (schema.required || []).forEach(function(k) {
            if (data[k] == null) throw geminiError('InvalidResponseError', 'Phản hồi AI thiếu thông tin cần thiết.');
        });
        Object.keys(schema.properties || {}).forEach(function(k) {
            if (data[k] != null) geminiValidateJson(data[k], schema.properties[k]);
        });
    } else if (type === 'ARRAY') data.forEach(function(item) { geminiValidateJson(item, schema.items); });
}
function geminiFailMessage(stats, calls) {
    if (!calls) return 'Các key/model đang tạm nghỉ do giới hạn lượt dùng hoặc quá tải. Hãy thử lại sau ít phút.';
    if (stats.quota && !stats.busy && !stats.other.length) return 'Gemini đang giới hạn lượt dùng. Hãy chờ rồi thử lại hoặc kiểm tra hạn mức API.';
    const parts = [];
    if (stats.invalid) parts.push('API key không hợp lệ');
    if (stats.quota) parts.push('đang giới hạn lượt dùng');
    if (stats.busy) parts.push('máy chủ quá tải');
    return 'Chưa nhận được kết quả (' + parts.concat(stats.other.slice(-2)).join('; ') + '). Câu đang viết vẫn được giữ, bạn có thể thử lại.';
}
async function geminiGenerateJson(prompt, schema, temperature, options) {
    options = options || {};
    const session = avgCardSession, started = Date.now();
    geminiEnsureActive(session, options);
    if (navigator.onLine === false) throw new Error('Thiết bị đang mất mạng. Hãy kết nối Internet rồi thử lại.');
    await avgLoadCardKeys();
    geminiEnsureActive(session, options);
    const keys = getGeminiCardKeys();
    if (!keys.length) throw new Error(AVG_NO_KEY_MSG);
    const stats = { invalid: 0, quota: 0, busy: 0, other: [] };
    let calls = 0;
    // Visit fallback models before spending all attempts on different keys for one model.
    outer: for (let n = 0; n < keys.length; n++) {
        const idx = (geminiKeyStart + n) % keys.length, key = keys[idx], kid = geminiKeyId(key);
        modelLoop: for (const m of GEMINI_TEXT_MODELS) {
            const h = geminiHealth();
            if ((h.dead[kid] || 0) > Date.now() || (h.busy[m.model] || 0) > Date.now()
                || (h.block[m.model + '|' + kid] || 0) > Date.now()) continue;
            const config = { responseMimeType: 'application/json', responseSchema: schema, maxOutputTokens: 6144 };
            if (/^gemini-2\./.test(m.model)) config.temperature = temperature;
            if (m.thinking) config.thinkingConfig = m.thinking;
            const body = JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], generationConfig: config });
            for (let attempt = 0; attempt < 2; attempt++) {
                geminiEnsureActive(session, options);
                const remaining = GEMINI_TOTAL_TIMEOUT_MS - (Date.now() - started);
                if (calls >= GEMINI_MAX_CALLS || remaining < 100) break outer;
                calls++;
                let retryMs = 800 * Math.pow(2, attempt) + Math.random() * 250;
                try {
                    const result = await geminiRequest('https://generativelanguage.googleapis.com/v1beta/models/' + m.model + ':generateContent?key=' + encodeURIComponent(key),
                        body, Math.min(GEMINI_TEXT_TIMEOUT_MS, remaining), session, options);
                    geminiEnsureActive(session, options);
                    const res = result.response, data = result.data;
                    if (res.ok) {
                        if (!data) throw geminiError('InvalidResponseError', 'Phản hồi không phải JSON.');
                        const candidate = data.candidates && data.candidates[0];
                        if ((data.promptFeedback && data.promptFeedback.blockReason)
                            || (candidate && /SAFETY|PROHIBITED|BLOCKLIST|RECITATION/.test(candidate.finishReason || ''))) {
                            throw geminiError('ContentBlockedError', 'AI chưa thể xử lý nội dung này. Hãy đổi cách diễn đạt rồi thử lại.');
                        }
                        const parts = (candidate && candidate.content && candidate.content.parts) || [];
                        const text = parts.filter(function(p) { return !p.thought; }).map(function(p) { return p.text || ''; }).join('');
                        let parsed;
                        try {
                            parsed = spParseJson(text);
                            geminiValidateJson(parsed, schema);
                            if (options.validate) options.validate(parsed);
                        } catch (e) { throw geminiError('InvalidResponseError', 'AI trả về dữ liệu chưa đầy đủ.'); }
                        geminiKeyStart = idx;
                        return parsed;
                    }
                    const err = await geminiReadError(res, data);
                    console.warn('[Gemini] ' + m.model + ' · HTTP ' + err.status + ' · ' + err.reason);
                    if ((err.status === 400 || err.status === 401) && /API_KEY|API key|UNAUTHENTICATED/i.test(err.reason + err.message)) {
                        stats.invalid++;
                        geminiMark('dead', kid, 86400000);
                        continue modelLoop;
                    }
                    if (err.status === 429) {
                        stats.quota++;
                        if (/PerDay|daily|quota_exceeded/i.test(err.quota + ' ' + err.reason) || err.retryMs > 5000) {
                            geminiMark('block', m.model + '|' + kid, Math.max(err.retryMs, /PerDay|daily/i.test(err.quota) ? 10800000 : 60000));
                            continue modelLoop;
                        }
                        retryMs = Math.max(retryMs, err.retryMs);
                        if (attempt) geminiMark('block', m.model + '|' + kid, 60000);
                    } else if (err.status === 408 || err.status >= 500) {
                        stats.busy++;
                        if (attempt) geminiMark('busy', m.model, 30000);
                    } else {
                        stats.other.push(m.model + ': HTTP ' + err.status);
                        if (err.status === 403 || err.status === 404) geminiMark('block', m.model + '|' + kid, 600000);
                        continue modelLoop;
                    }
                } catch (e) {
                    if (e.name === 'CancelledError' || e.name === 'ContentBlockedError') throw e;
                    if (navigator.onLine === false) throw new Error('Kết nối mạng bị gián đoạn. Câu đang viết vẫn được giữ; hãy kết nối lại rồi thử.');
                    if (e.name === 'TimeoutError' || e.name === 'AbortError') {
                        stats.other.push(m.model + ': kết nối quá thời gian chờ');
                        continue modelLoop;
                    }
                    stats.other.push(e.name === 'InvalidResponseError' ? m.model + ': phản hồi chưa đầy đủ' : 'kết nối mạng gián đoạn');
                }
                if (attempt === 0 && calls < GEMINI_MAX_CALLS && Date.now() - started + retryMs < GEMINI_TOTAL_TIMEOUT_MS) {
                    if (options.onRetry) options.onRetry();
                    await geminiWait(retryMs, session, options);
                } else break;
            }
        }
    }
    geminiEnsureActive(session, options);
    throw new Error(geminiFailMessage(stats, calls));
}

function spEsc(s) {
    // AI hay viết dấu gạch dài: đổi sang dấu phẩy cho dễ đọc
    return String(s == null ? '' : s).replace(/\s*—\s*/g, ', ').replace(/[&<>"']/g, function(c) {
        return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
}

    } catch (e) { /* keep the in-memory copy when storage is unavailable */ }
}

function spNewLog() {
    return { v: 1, unlocked: 1, passed: [false, false, false], best: {}, mistakes: [], usage: null, mnemonic: '', attempts: 0, updated: 0 };
}
function spNormalizeLog(raw) {
    const log = spNewLog();
    if (!raw || typeof raw !== 'object') return log;
    log.unlocked = Math.min(3, Math.max(1, parseInt(raw.unlocked, 10) || 1));
    if (Array.isArray(raw.passed)) log.passed = [0, 1, 2].map(function(i) { return !!raw.passed[i]; });
    if (raw.best && typeof raw.best === 'object') log.best = raw.best;
    if (Array.isArray(raw.mistakes)) log.mistakes = raw.mistakes.filter(function(m) { return m && m.tag; });
    log.usage = raw.usage || null;
    log.mnemonic = raw.mnemonic || '';
    log.attempts = parseInt(raw.attempts, 10) || 0;
    log.updated = parseInt(raw.updated, 10) || 0;
    return log;
}
function spSortMistakes(list) {
    return list.sort(function(x, y) { return (y.n || 1) - (x.n || 1) || (y.at || 0) - (x.at || 0); });
}
// Gộp lịch sử từ field (đồng bộ từ máy tính) với lịch sử lưu trong máy này
function spMergeLogs(a, b) {
    a = spNormalizeLog(a); b = spNormalizeLog(b);
    const m = spNewLog();
    m.unlocked = Math.max(a.unlocked, b.unlocked);
    m.passed = [0, 1, 2].map(function(i) { return a.passed[i] || b.passed[i]; });
    Object.keys(Object.assign({}, a.best, b.best)).forEach(function(k) {
        const x = a.best[k], y = b.best[k];
        m.best[k] = !x ? y : !y ? x : ((y.at || 0) > (x.at || 0) ? y : x);
    });
    const byTag = {};
    a.mistakes.concat(b.mistakes).forEach(function(x) {
        const cur = byTag[x.tag];
        if (!cur) { byTag[x.tag] = Object.assign({}, x); return; }
        cur.n = Math.max(cur.n || 1, x.n || 1);
        if ((x.at || 0) > (cur.at || 0)) { cur.at = x.at; cur.label = x.label || cur.label; }
    });
    m.mistakes = spSortMistakes(Object.values(byTag)).slice(0, 8);
    m.usage = a.usage || b.usage;
    m.mnemonic = a.mnemonic || b.mnemonic;
    m.attempts = Math.max(a.attempts, b.attempts);
    m.updated = Math.max(a.updated, b.updated);
    return m;
}

    const st = spState;
    if (!st) return;
    st.log.updated = Date.now();
    spStoreSet(st.storeKey, st.log);
    if (hasAnkiBridge()) {
        try {

            });
        } catch (e) { console.warn('[Luyện câu] pycmd lỗi:', e); }
    }
}
function spRecordMistakes(issues) {
    const st = spState, now = Date.now();
    const weak = spStoreGet('sp_weak_v1') || {};
    issues.forEach(function(i) {
        // Chỉ ghi lỗi ngôn ngữ (bỏ lỗi "sai đề"), mỗi đề chỉ đếm một lần cho mỗi loại lỗi
        if (i.severity !== 'error' || i.category === 'task' || st.taskTags[i.tag]) return;
        st.taskTags[i.tag] = true;
        const label = i.label_vi || i.fix || i.tag;
        const cur = st.log.mistakes.find(function(m) { return m.tag === i.tag; });
        if (cur) { cur.n = (cur.n || 1) + 1; cur.at = now; cur.label = label; }
        else st.log.mistakes.push({ tag: i.tag, label: label, cat: i.category, n: 1, at: now });
        const w = weak[i.tag] || { label: label, cat: i.category, n: 0 };
        w.n += 1; w.at = now; w.label = label;
        weak[i.tag] = w;
    });
    st.log.mistakes = spSortMistakes(st.log.mistakes).slice(0, 8);
    // Điểm yếu chung (mọi từ) chỉ lưu trong máy này, giữ 20 loại hay mắc nhất
    const top = Object.entries(weak).sort(function(a, b) { return b[1].n - a[1].n || (b[1].at || 0) - (a[1].at || 0); }).slice(0, 20);
    spStoreSet('sp_weak_v1', Object.fromEntries(top));
}

function spCleanUsage(u) {
    if (!u || typeof u !== 'object') return null;
    const pick = function(arr) {
        return (Array.isArray(arr) ? arr : []).filter(function(it) { return it && it.form; }).slice(0, 5).map(function(it) {
            return { form: String(it.form), note_vi: String(it.note_vi || ''), example: String(it.example || '') };
        });
    };
    const out = { particles: pick(u.particles), collocations: pick(u.collocations), structures: pick(u.structures), phrases: pick(u.phrases) };
    return (out.particles.length + out.collocations.length + out.structures.length + out.phrases.length) ? out : null;
}

        + ' Keep it short (3–6 slots) and built around the grammar point this task trains (aspect, complement, measure word, word order, 把/被/比), not one slot per word.'
        + ' Only the target word and grammar function words (的, 地, 得, 是, 很, 把, 被, 比, 在, 不, 没) may be written literally; every other word of the answer, especially the word that collocates with the target word, must stay a slot label so the formula never gives the answer away.'

        '- "structure_vi": max 20 Vietnamese words naming the pattern and what each unusual slot means, without giving the answer, e.g. '
            + (zh ? '"Câu chữ 把: đưa tân ngữ lên trước động từ, sau động từ cần bổ ngữ".' : '"Cụm danh từ: mạo từ + tính từ đứng trước danh từ".'),
        '- "structure_breakdown": the FIRST reference answer split into the slots of "structure", in order, one part per slot; the parts joined must give back that answer (it stays hidden until the learner finishes). "slot": the slot label exactly as in the formula; "text": the exact words filling it' + (zh ? ' (Hanzi only)' : '') + '; "note_vi": max 12 Vietnamese words for a beginner who knows no grammar terms, saying WHY this word/form goes there (e.g. '
            + (zh ? '"书 luôn dùng lượng từ 本", "了 sau động từ: việc đã làm xong"' : '"rely luôn đi với giới từ on"') + ').',

    );
    if (needUsage) parts.push(spUsageInstruction());
    parts.push('Return JSON only.');
    return parts.join('\n');
}

function spNormalizeBreakdown(list) {
    return (Array.isArray(list) ? list : []).filter(function(p) { return p && p.text; }).slice(0, 8).map(function(p) {
        return { slot: String(p.slot || '').trim(), text: String(p.text), note_vi: String(p.note_vi || '') };
    });
}
function spNormalizeGrade(d) {
    if (!d || typeof d !== 'object') throw new Error('AI trả về dữ liệu không hợp lệ');
    const issues = (Array.isArray(d.issues) ? d.issues : []).filter(function(i) { return i && (i.hint_vi || i.fix); }).slice(0, 5).map(function(i) {
        return {
            severity: i.severity === 'suggestion' ? 'suggestion' : 'error',
            category: String(i.category || ''),
            excerpt: String(i.excerpt || ''),
            hint_vi: String(i.hint_vi || ''),
            fix: String(i.fix || ''),
            explain_vi: String(i.explain_vi || ''),
            tag: String(i.tag || i.category || 'other').toLowerCase().slice(0, 60),
            label_vi: String(i.label_vi || '')
        };
    });
    const str = function(v) { return v == null ? '' : String(v); };
    return {
        // Không tin hẳn cờ pass: còn lỗi nặng thì chưa đạt
        pass: d.pass === true && !issues.some(function(i) { return i.severity === 'error'; }),
        issues: issues,
        summary_vi: str(d.summary_vi), corrected: str(d.corrected), corrected_pinyin: str(d.corrected_pinyin),
        model_answer: str(d.model_answer), model_pinyin: str(d.model_pinyin),
        explain_vi: str(d.explain_vi), upgrade_vi: str(d.upgrade_vi),
        structure: str(d.structure).trim(),
        structure_breakdown: spNormalizeBreakdown(d.structure_breakdown),
        structure_explain_vi: str(d.structure_explain_vi),
        raw: d
    };
}

// ---------- Giao diện ----------
function spDefaultLevel(log) {
    for (let l = 1; l <= log.unlocked; l++) if (!log.passed[l - 1]) return l;
    return log.unlocked;
}
function spSaveDraft() {
    const st = spState;
    if (!st || !spIsCurrent(st.token)) return;
    spStoreSet(st.draftKey, { v: 1, mode: st.mode, task: st.task, answer: st.input.value,
        attempts: st.attempts, finished: st.finished, lastGrade: st.lastGrade, taskTags: st.taskTags, at: Date.now() });
}
function spRestoreDraft(st) {
    const d = spStoreGet(st.draftKey);
    if (!d || d.v !== 1 || Date.now() - d.at > 86400000 || !SP_LEVELS[String(d.mode)]
        || (d.mode !== 'free' && d.mode > st.log.unlocked)) return;
    if (d.task && (typeof d.task.task_vi !== 'string' || !Array.isArray(d.task.reference_answers)
        || !d.task.reference_answers.length || !Array.isArray(d.task.must_use))) return;
    st.mode = d.mode;
    st.task = d.task || null;
    st.input.value = typeof d.answer === 'string' ? d.answer : '';
    st.attempts = Math.max(0, Math.min(SP_MAX_ATTEMPTS, Number(d.attempts) || 0));
    st.finished = !!d.finished;
    st.lastGrade = d.lastGrade && Array.isArray(d.lastGrade.issues) ? d.lastGrade : null;
    st.taskTags = d.taskTags && typeof d.taskTags === 'object' ? d.taskTags : {};
}
function spInit() {
    const input = spEl('sp-input'), root = spEl('sp-container');
    if (!input || !root) return;
    if (spState && spState.root === root && spState.input === input) return;
    const word = spText('.word');
    if (!word) return;
    const fieldLog = spJson('#user-sentence-data');
    const storeKey = 'sp_log_v1::' + word.toLowerCase();
    const log = spMergeLogs(fieldLog && fieldLog.v ? fieldLog : null, spStoreGet(storeKey));
    spState = {
        token: Math.random().toString(36).slice(2), word: word, lang: /[㐀-鿿]/.test(word) ? 'zh' : 'en',
        root: root, input: input, draftKey: 'sp_draft_v1::' + word.toLowerCase() + '::' + spText('.definition-vi'),
        storeKey: storeKey, log: log, mode: spDefaultLevel(log), error: '', request: null,
        task: null, attempts: 0, finished: false, lastGrade: null, taskTags: {}, busy: false
    };
    spRestoreDraft(spState);
    input.addEventListener('keydown', function(e) {
        if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) { e.preventDefault(); spCheck(); }
    });
    input.addEventListener('input', spSaveDraft);
    input.addEventListener('focus', spOnInputFocus);
    root.addEventListener('click', function(e) {
        const btn = e.target.closest('[data-sp-say]');
        if (btn) speakTextWithEdge(btn.getAttribute('data-sp-say'), e);
    });
    spRenderLevels();
    spRenderMemory();
    spRenderTask();
    spUpdateButtons();
    if (spState.lastGrade || spState.finished) spRenderResult({ gaveUp: !spState.lastGrade });
}
function spStartRequest(kind, token) {
    const request = { kind: kind, startedAt: Date.now(), cancelled: false,
        isCurrent: function() { return spIsCurrent(token); },
        onRetry: function() {
            if (spIsCurrent(token)) spSetBusy(true, 'Kết nối chưa ổn định, đang thử lại...');
        }
    };
    spState.request = request;
    return request;
}
function spCancelRequest(message) {
    const st = spState;
    if (!st || !st.busy) return;
    spSaveDraft();
    if (st.request) {
        st.request.cancelled = true;
        if (st.request.cancel) st.request.cancel();
    }
    // Invalidate the operation even if AbortController is unavailable on the device.
    st.token = Math.random().toString(36).slice(2);
    st.request = null;
    st.error = message || 'Đã dừng yêu cầu. Câu đang viết vẫn được giữ; bạn có thể thử lại.';
    spSetBusy(false);
}
function spSetBusy(busy, text) {
    const st = spState, box = spEl('sp-status');
    if (!st || !box) return;
    st.busy = busy;
    if (busy) {
        st.error = '';
        box.innerHTML = '<span class="sp-spinner"></span><span>' + spEsc(text || 'Đang xử lý...') + '</span>'
            + '<button type="button" class="sp-btn tappable sp-btn-cancel" onclick="spCancelRequest()">Dừng</button>';
        box.style.display = 'flex';
    } else if (st.error) {
        box.textContent = st.error;
        box.style.display = 'flex';
    } else {
        box.style.display = 'none';
        box.innerHTML = '';
    }
    spUpdateButtons();
}
function spRenderLevels() {
    const st = spState;
    let html = '';
    [1, 2, 3].forEach(function(l) {
        const locked = l > st.log.unlocked, passed = st.log.passed[l - 1];
        html += '<button type="button" class="sp-level tappable' + (st.mode === l ? ' active' : '') + (locked ? ' locked' : '') + '"'
            + (locked ? ' disabled title="Đạt bậc trước để mở"' : '') + ' onclick="spSetMode(' + l + ')">'
            + (passed ? '✅' : locked ? '🔒' : '⭐') + ' ' + SP_LEVELS[l].name + '<small>' + SP_LEVELS[l].short + '</small></button>';
    });
    html += '<button type="button" class="sp-level tappable' + (st.mode === 'free' ? ' active' : '') + '" onclick="spSetMode(\'free\')">✍️ '
        + SP_LEVELS.free.name + '<small>' + SP_LEVELS.free.short + '</small></button>';
    spEl('sp-levels').innerHTML = html;
}
function spRenderMemory() {
    const st = spState, box = spEl('sp-memory');
    const best = st.log.best['3'] || st.log.best['2'] || st.log.best['1'] || st.log.best.free;
    const mistakes = st.log.mistakes.slice(0, 3);
    if (!best && !mistakes.length) { box.style.display = 'none'; box.innerHTML = ''; return; }
    let html = '';
    if (best) {
        html += '<div class="sp-mem-label">🏆 Câu tốt nhất của bạn (' + (SP_LEVELS[best.level] || SP_LEVELS.free).name + ')</div>'
            + '<div class="sp-mem-text">' + spEsc(best.text) + ' ' + spSayBtn(best.text) + '</div>';
    }
    if (mistakes.length) {
        html += '<div class="sp-mem-mistakes"><div class="sp-mem-label">⚠️ Lỗi bạn hay mắc với từ này</div><ul>'
            + mistakes.map(function(m) { return '<li>' + spEsc(m.label) + (m.n > 1 ? ' <span class="sp-count">×' + m.n + '</span>' : '') + '</li>'; }).join('')
            + '</ul></div>';
    }
    box.innerHTML = html;
    box.style.display = 'block';
}
function spRenderTask() {
    const st = spState, box = spEl('sp-task'), input = spEl('sp-input');
    // Không khoá ô nhập (trên điện thoại ô bị khoá thì chạm vào không hiện bàn phím);
    // chạm vào ô lúc chưa có đề / đã xong sẽ tự ra đề tiếp (xem spOnInputFocus)
    input.placeholder = st.finished
        ? (st.mode === 'free' ? 'Chạm để viết câu khác...' : 'Chạm để làm đề tiếp theo...')
        : (st.mode !== 'free' && !st.task ? 'Chạm vào đây để AI ra đề, rồi viết câu trả lời...'
            : st.lang === 'zh' ? 'Viết bằng chữ Hán (chưa có bộ gõ thì gõ pinyin cũng được)...' : 'Viết câu trả lời của bạn ở đây...');
    if (st.mode === 'free') {
        box.innerHTML = '<div class="sp-task-label">✍️ Tự do</div><div class="sp-task-text">Tự đặt một câu bất kỳ có dùng "<b>'
            + spEsc(st.word) + '</b>". Câu và lỗi vẫn được lưu nhưng không tính lên bậc.</div>';
        return;
    }
    const label = SP_LEVELS[String(st.mode)].name + ' · ' + SP_LEVELS[String(st.mode)].short;
    if (!st.task) {
        box.innerHTML = '<div class="sp-task-label">' + label + '</div><div class="sp-task-text sp-muted">Bấm <b>🎲 Đề mới</b> hoặc chạm vào ô bên dưới để AI ra đề.</div>';
        return;
    }
    const t = st.task;
    const progress = st.finished ? 'Đã xong' : 'Lần thử ' + (st.attempts + 1) + '/' + SP_MAX_ATTEMPTS;
    box.innerHTML = '<div class="sp-task-label">' + label + ' · ' + progress + '</div>'
        + spTaskTextHtml(t)
        + (t.focus_hint_vi ? '<div class="sp-task-hint">💡 ' + spFmt(t.focus_hint_vi) + '</div>' : '')
        + (t.structure ? '<div class="sp-structure"><div class="sp-structure-label">🧩 Cấu trúc gợi ý</div><div class="sp-structure-pattern">'
            + spEsc(t.structure) + '</div>' + (t.structure_vi ? '<div>' + spFmt(t.structure_vi) + '</div>' : '') + spLegendHtml(t.structure) + '</div>' : '')
        + (t.must_use.length ? '<div class="sp-task-must">Phải dùng: ' + t.must_use.map(function(w) { return '<code>' + spEsc(w) + '</code>'; }).join(' ') + '</div>' : '');

// Bậc 1: hiện rõ CỤM cần dịch; câu dài (nếu có) chỉ là ngữ cảnh, tô màu đúng cụm đó trong câu
function spTaskTextHtml(t) {
    const phrase = t.phrase_vi;
    const lang = spState.lang === 'zh' ? 'tiếng Trung' : 'tiếng Anh';
    if (!phrase) {
        // Bậc 2: dịch cả câu; Bậc 3: trả lời tình huống
        const lead = spState.mode === 2 ? 'Viết câu ' + lang + ' cho:'
            : spState.mode === 3 ? 'Trả lời bằng 1–2 câu ' + lang + ':' : '';
        return (lead ? '<div class="sp-task-lead">' + lead + '</div>' : '') + '<div class="sp-task-text">' + spFmt(t.task_vi) + '</div>';
    }
    let html = '<div class="sp-task-text">Viết cụm ' + lang + ' cho:</div><div class="sp-phrase">' + spEsc(phrase) + '</div>';
    const context = (t.task_vi || '').trim();
    // Chỉ hiện ngữ cảnh khi là câu thật (không phải câu lệnh "Hãy dịch..." hay chỉ lặp lại cụm)
    const rest = context.toLowerCase().replace(phrase.toLowerCase(), '').replace(/[\s.,:;!?"'“”]+/g, ' ').trim();
    const isInstruction = /(hãy|dịch|cụm từ sau|sang tiếng)/i.test(rest);
    if (context && rest.split(' ').length >= 3 && !isInstruction) {
        const pos = context.toLowerCase().indexOf(phrase.toLowerCase());
        const marked = pos >= 0
            ? spEsc(context.slice(0, pos)) + '<mark class="sp-phrase-mark">' + spEsc(context.slice(pos, pos + phrase.length)) + '</mark>' + spEsc(context.slice(pos + phrase.length))
            : spEsc(context);
        html += '<div class="sp-context">Ngữ cảnh: ' + marked + '</div>';
    }
    return html;
}
function spOnInputFocus() {
    const st = spState, input = spEl('sp-input');
    if (!st) return;
    // Bàn phím điện thoại hay che mất ô nhập -> cuộn ô lên giữa màn hình
    // (iPhone tự cuộn ô lên; chỉ cuộn thêm khi ô thật sự nằm dưới mép bàn phím, tránh giật màn hình)
    setTimeout(function() {
        try {
            const visibleBottom = window.visualViewport ? window.visualViewport.height : window.innerHeight;
            if (input === spEl('sp-input') && input.getBoundingClientRect().bottom > visibleBottom) input.scrollIntoView({ block: 'center' });
        } catch (e) { /* trình duyệt cũ */ }
    }, 400);
    if (st.busy) return;
    if (st.finished) {
        if (st.mode === 'free') spResetFree();
        else spNewTask();
    } else if (st.mode !== 'free' && !st.task) {
        spNewTask();
    }
}
function spUpdateButtons() {
    const st = spState;
    const btnNew = spEl('sp-btn-new'), btnCheck = spEl('sp-btn-check'), btnReveal = spEl('sp-btn-reveal');
    btnNew.style.display = st.mode === 'free' ? 'none' : '';
    btnNew.disabled = st.busy;
    btnNew.textContent = st.task ? '🎲 Đề khác' : '🎲 Đề mới';
    btnCheck.disabled = st.busy || st.finished || (st.mode !== 'free' && !st.task);
    btnCheck.textContent = st.attempts > 0 && !st.finished ? '🔁 Chấm lại' : '✅ Chấm';
    const canReveal = !st.finished && ((st.task && st.task.reference_answers.length) || st.lastGrade);
    btnReveal.disabled = st.busy || !canReveal;
}
function spIssueHtml(i, revealed) {
    const cat = (spState && spState.lang === 'zh' && SP_CATEGORY_NAMES_ZH[i.category]) || SP_CATEGORY_NAMES[i.category] || i.category || 'Lỗi';
    let excerpt = '';
    if (i.excerpt) excerpt = '“' + spEsc(i.excerpt) + '”' + (revealed && i.fix ? ' → <b>' + spEsc(i.fix) + '</b>' : '');
    else if (revealed && i.fix) excerpt = '→ <b>' + spEsc(i.fix) + '</b>';
    return '<div class="sp-issue ' + (i.severity === 'error' ? 'error' : 'suggestion') + '">'
        + '<div class="sp-issue-cat">' + (i.severity === 'error' ? '' : 'Gợi ý · ') + spEsc(cat) + '</div>'
        + (excerpt ? '<div class="sp-issue-excerpt">' + excerpt + '</div>' : '')
        + '<div class="sp-issue-hint">' + spFmt(revealed ? (i.explain_vi || i.hint_vi) : i.hint_vi) + '</div>'
        + '</div>';
}
function spAnswersHtml(g, t) {
    const row = function(label, text, pinyin) {
        if (!text) return '';
        return '<div class="sp-answer"><div class="sp-answer-label">' + label + '</div><div class="sp-answer-text">' + spEsc(text) + ' ' + spSayBtn(text) + '</div>'
            + (pinyin ? '<div class="item-content-pinyin">' + spEsc(pinyin) + '</div>' : '') + '</div>';
    };
    let html = '';
    if (g && !g.pass && g.corrected) html += row('✏️ Câu của bạn sau khi sửa', g.corrected, g.corrected_pinyin);
    if (g && g.model_answer) {
        html += row(g.pass ? '✨ Câu mẫu tham khảo' : '✅ Câu mẫu', g.model_answer, g.model_pinyin);
    } else if (t && t.reference_answers.length) {
        html += t.reference_answers.map(function(a, k) { return row(k ? '🔁 Cách nói khác' : '✅ Đáp án', a, k ? '' : t.reference_pinyin); }).join('');
    }
    return html;
}
// Tên tiếng Việt của một ô trong công thức ("V" -> động từ), '' nếu là chữ thường (need to, it...)
function spSlotInfo(slot) {
    // Bỏ pinyin trong ngoặc ("把(bǎ)" -> "把"), kể cả ngoặc toàn khổ（）
    const key = String(slot || '').trim().toLowerCase().replace(/[(（][^)）]*[)）]/g, '').trim();
    const names = spState && spState.lang === 'zh' ? SP_SLOT_NAMES_ZH : SP_SLOT_NAMES;
    if (names[key]) return names[key];
    // "V1", "V2" (一边 V1 一边 V2) -> giải thích như "V"; giữ nguyên v2/v3 của tiếng Anh vì đã có ở trên
    const bare = key.replace(/\d+$/, '');
    if (bare !== key && names[bare]) return names[bare];
    if (spState && key && key.indexOf(spState.word.toLowerCase()) >= 0) return ['từ bạn đang học', 'giữ đúng nghĩa trên thẻ'];
    return null;
}
// Bảng "Ký hiệu" dưới công thức: chỉ giải thích những ký hiệu có trong công thức, theo thứ tự xuất hiện
function spLegendHtml(formula) {
    const seen = {}, rows = [];
    String(formula || '').split(/\s*\+\s*/).forEach(function(token) {
        // Bỏ pinyin trong ngoặc trước khi tách, để "把 (bǎ)" không hiện hai lần
        token = token.replace(/[(（][^)）]*[)）]/g, '').trim();
        // "for sb" -> giải thích "sb"; "a/the" giữ nguyên
        [token].concat(token.indexOf(' ') > 0 ? token.split(/\s+/) : []).forEach(function(part) {
            const info = spSlotInfo(part), key = part.toLowerCase();
            if (!info || seen[key] || info[0] === 'từ bạn đang học') return;
            seen[key] = true;
            rows.push('<div><b>' + spEsc(part) + '</b> = ' + spEsc(info[0]) + ': ' + spEsc(info[1]) + '</div>');
        });
    });
    return rows.length ? '<div class="sp-legend"><div class="sp-legend-title">📖 Ký hiệu</div>' + rows.join('') + '</div>' : '';
}
// Sau khi xong: tách câu đúng theo từng ô của công thức, giải thích vì sao từ đó đứng ở đó
function spStructureHtml(g, t) {
    // Có chấm thì dùng câu AI đã chấm; bấm xem đáp án luôn thì dùng phần tách sẵn lúc ra đề
    const src = g && g.structure_breakdown && g.structure_breakdown.length ? g : t;
    const formula = (src && src.structure) || (g && g.structure) || (t && t.structure) || '';
    const parts = (src && src.structure_breakdown) || [];
    const explain = (g && g.structure_explain_vi) || (t && t.structure_vi) || '';
    if (!formula && !parts.length && !explain) return '';
    let html = '<div class="sp-breakdown"><div class="sp-structure-label">🧩 Giải thích cấu trúc</div>';
    if (formula) html += '<div class="sp-structure-pattern">' + spEsc(formula) + '</div>';
    if (parts.length) {
        // Dòng ghép: V = rely + prep = on + the + N = market mechanism
        html += '<div class="sp-bd-map">' + parts.map(function(p) {
            return p.slot && p.slot.toLowerCase() !== p.text.toLowerCase()
                ? '<b>' + spEsc(p.slot) + '</b> = ' + spEsc(p.text) : '<b>' + spEsc(p.text) + '</b>';
        }).join(' + ') + '</div>';
        html += '<div class="sp-bd-parts">' + parts.map(function(p) {
            const info = spSlotInfo(p.slot);
            const name = info ? '<i>' + spEsc(info[0]) + '</i>' : '';
            return '<div class="sp-bd-row"><span class="sp-bd-slot">' + spEsc(p.slot) + '</span><span class="sp-bd-text">' + spEsc(p.text) + '</span>'
                + (name || p.note_vi ? '<span class="sp-bd-note">' + name + (name && p.note_vi ? ': ' : '') + spFmt(p.note_vi) + '</span>' : '') + '</div>';
        }).join('') + '</div>';
    } else {
        html += spLegendHtml(formula);
    }
    if (explain) html += '<div class="sp-bd-explain">📝 ' + spFmt(explain) + '</div>';
    return html + '</div>';
}
function spUsageHtml(u) {
    const sec = function(title, items) {
        if (!items || !items.length) return '';
        return '<div class="sp-usage-sec"><div class="sp-usage-title">' + title + '</div>' + items.map(function(it) {
            return '<div class="sp-usage-item"><b>' + spEsc(it.form) + '</b>' + (it.note_vi ? ': ' + spEsc(it.note_vi) : '')
                + (it.example ? '<div class="sp-usage-ex">' + spEsc(it.example) + '</div>' : '') + '</div>';
        }).join('') + '</div>';
    };
    return '<div class="sp-usage"><div class="sp-usage-head">📇 Thẻ tóm tắt cách dùng "' + spEsc(spState.word) + '"</div>'
        + sec(spState.lang === 'zh' ? '🔗 Lượng từ / trợ từ / bổ ngữ đi kèm' : '🔗 Giới từ / tiểu từ đi kèm', u.particles)
        + sec(spState.lang === 'zh' ? '🤝 Từ hay đi kèm (搭配)' : '🤝 Collocation', u.collocations)
        + sec('🏗️ Cấu trúc', u.structures)
        + sec(spState.lang === 'zh' ? '🚀 Thành ngữ / cụm cố định' : '🚀 Phrasal verb / idiom / cụm cố định', u.phrases)
        + '</div>';
}
function spMoreHtml(g) {

    let html = '';
    if (g && g.explain_vi) html += '<div class="sp-explain">📝 ' + spFmt(g.explain_vi) + '</div>';

    if (st.log.mnemonic) html += '<div class="sp-mnemonic">🧠 <b>Mẹo nhớ:</b> ' + spFmt(st.log.mnemonic) + '</div>';

}
function spRenderResult(opts) {
    opts = opts || {};
    const st = spState, g = st.lastGrade, t = st.task;
    const revealed = st.finished;
    let html = '';
    if (g) {
        let head;
        if (g.pass) head = '🎉 Đạt!';
        else if (opts.gaveUp) head = '📘 Đáp án';
        else if (revealed) head = '📘 Chưa đạt sau ' + st.attempts + ' lần, xem đáp án bên dưới';
        else head = '🔁 Chưa đạt, tự sửa lại nhé (còn ' + (SP_MAX_ATTEMPTS - st.attempts) + ' lần)';
        html += '<div class="sp-verdict ' + (g.pass ? 'pass' : 'fail') + '"><div class="sp-verdict-head">' + head + '</div>'
            + (g.summary_vi ? '<div>' + spFmt(g.summary_vi) + '</div>' : '') + '</div>';
        if (opts.unlockedNow) html += '<div class="sp-unlock">🔓 Đã mở ' + SP_LEVELS[opts.unlockedNow].name + ': ' + SP_LEVELS[opts.unlockedNow].short + '!</div>';
        if (opts.allDone) html += '<div class="sp-unlock">🏆 Bạn đã vượt cả 3 bậc với từ này!</div>';
        html += g.issues.filter(function(i) { return i.severity === 'error'; }).map(function(i) { return spIssueHtml(i, revealed); }).join('');
        if (revealed) html += g.issues.filter(function(i) { return i.severity !== 'error'; }).map(function(i) { return spIssueHtml(i, true); }).join('');
    } else if (opts.gaveUp) {
        html += '<div class="sp-verdict fail"><div class="sp-verdict-head">📘 Đáp án</div></div>';
    }
    if (revealed) {
        html += spAnswersHtml(g, t);
        html += spStructureHtml(g, t);
        const more = spMoreHtml(g);
        if (more) {
            html += '<button type="button" class="toggle-section-btn" onclick="toggleSection(\'sp-more\')">📖 Xem thêm: giải thích, thẻ cách dùng, mẹo nhớ ▼</button>'
                + '<div id="sp-more" class="collapsible-section" style="display: none;">' + more + '</div>';
        }
        html += st.mode === 'free'
            ? '<button type="button" class="sp-btn tappable sp-btn-next" onclick="spResetFree()">✍️ Viết câu khác</button>'
            : '<button type="button" class="sp-btn tappable sp-btn-next" onclick="spNewTask()">🎲 Đề tiếp theo</button>';
    }
    spEl('sp-result').innerHTML = html;
}
function spResetTask() {
    const st = spState;
    st.attempts = 0; st.finished = false; st.lastGrade = null; st.taskTags = {}; st.error = '';
    const status = spEl('sp-status');
    if (status && !st.busy) { status.style.display = 'none'; status.innerHTML = ''; }
    spEl('sp-input').value = '';
    spEl('sp-result').innerHTML = '';
}

function spSetMode(mode) {
    const st = spState;
    if (!st || st.busy || st.mode === mode) return;
    if (mode !== 'free' && mode > st.log.unlocked) return;
    st.mode = mode;
    st.task = null;
    spResetTask();
    spRenderLevels();
    spRenderTask();
    spUpdateButtons();
    spSaveDraft();
    if (mode !== 'free') spNewTask();
}
function spResetFree() {
    const st = spState;
    if (!st || st.busy) return;
    spResetTask();
    spRenderTask();
    spUpdateButtons();
    spSaveDraft();
    spEl('sp-input').focus();
}
async function spNewTask() {
    const st = spState;
    if (!st || st.busy || st.mode === 'free') return;
    const token = st.token, level = st.mode, needUsage = !st.log.usage;
    const textAtStart = spEl('sp-input').value;
    const request = spStartRequest('task', token);
    request.validate = function(d) {
        if (!d.task_vi.trim() || !d.reference_answers.some(function(s) { return s.trim(); })) throw new Error('Đề bài chưa đầy đủ');
    };
    spSetBusy(true, 'Đang ra đề ' + SP_LEVELS[String(level)].name + '...');
    try {
        const data = await geminiGenerateJson(spBuildTaskPrompt(level, needUsage), spSchema(SP_TASK_SCHEMA, needUsage), 0.9, request);
        if (!spIsCurrent(token)) return;
        if (!data || !data.task_vi) throw new Error('AI không trả về đề bài');
        const list = function(v) { return Array.isArray(v) ? v.map(String).filter(Boolean) : []; };
        st.task = {
            task_vi: String(data.task_vi),
            // Bậc 1: chỉ dịch đúng cụm này (AI hay đưa cả câu dài làm người học không biết dịch phần nào)
            phrase_vi: level === 1 ? String(data.phrase_vi || '').trim() : '',
            focus: String(data.focus || ''),
            focus_hint_vi: String(data.focus_hint_vi || ''),
            structure: String(data.structure || '').trim(),
            structure_vi: String(data.structure_vi || '').trim(),
            structure_breakdown: spNormalizeBreakdown(data.structure_breakdown),   // giấu tới khi xong / xem đáp án
            // Bậc 1–2 chỉ hiện từ đang học: AI hay liệt kê cả từ đi kèm, làm lộ đáp án
            must_use: level === 3 && list(data.must_use).length ? list(data.must_use).slice(0, 3) : [st.word],
            reference_answers: list(data.reference_answers),
            reference_pinyin: String(data.reference_pinyin || '')
        };
        if (needUsage) { spStoreUsage(data); if (st.log.usage) spSaveLog(); }
        spRememberTask(st.task.task_vi);
        // Người học đã gõ trong lúc chờ đề -> giữ lại chữ đó, không xoá
        const typed = spEl('sp-input').value;
        spResetTask();
        if (typed !== textAtStart) spEl('sp-input').value = typed;
        spRenderTask();
        spEl('sp-input').focus();
    } catch (e) {
        if (spIsCurrent(token)) { st.error = e.message; console.warn('[Luyện câu]', e.message); showPracticeMessage('❌ ' + e.message, 'error'); }
    } finally {
        if (spIsCurrent(token)) { st.request = null; spSetBusy(false); spSaveDraft(); }
    }
}
async function spCheck() {
    const st = spState;
    if (!st || st.busy || st.finished) return;
    if (st.mode !== 'free' && !st.task) return;
    const answer = spEl('sp-input').value.trim();
    if (!answer) { showPracticeMessage('⚠️ Hãy viết câu trả lời trước khi chấm', 'error'); return; }
    const token = st.token, attempt = st.attempts + 1, needUsage = !st.log.usage;
    const request = spStartRequest('grade', token);
    spSetBusy(true, 'Đang chấm...');
    try {
        const g = spNormalizeGrade(await geminiGenerateJson(spBuildGradePrompt(answer, attempt, needUsage), spSchema(SP_GRADE_SCHEMA, needUsage), 0.2, request));
        if (!spIsCurrent(token)) return;
        st.attempts = attempt;
        st.lastGrade = g;
        st.log.attempts += 1;
        if (needUsage) spStoreUsage(g.raw);
        spRecordMistakes(g.issues);
        const opts = {};
        if (g.pass) {
            st.finished = true;
            st.log.best[String(st.mode)] = { level: String(st.mode), text: answer, at: Date.now() };
            if (st.mode !== 'free') {
                const wasPassed = st.log.passed[st.mode - 1];
                st.log.passed[st.mode - 1] = true;
                if (st.mode < 3 && st.log.unlocked < st.mode + 1) { st.log.unlocked = st.mode + 1; opts.unlockedNow = st.mode + 1; }
                opts.allDone = !wasPassed && st.log.passed.every(Boolean);
            }
        } else if (attempt >= SP_MAX_ATTEMPTS) {
            st.finished = true;
        }
        spSaveLog();
        spRenderResult(opts);
        spRenderLevels();
        spRenderMemory();
        spRenderTask();
    } catch (e) {
        if (spIsCurrent(token)) { st.error = e.message; console.warn('[Luyện câu]', e.message); showPracticeMessage('❌ ' + e.message, 'error'); }
    } finally {
        if (spIsCurrent(token)) { st.request = null; spSetBusy(false); spSaveDraft(); }
    }
}
function spReveal() {
    const st = spState;
    if (!st || st.busy || st.finished) return;
    st.finished = true;
    spRenderResult({ gaveUp: true });
    spRenderTask();
    spUpdateButtons();
    spSaveDraft();
}

// Chuẩn hoá dữ liệu từ vựng (bản do thẻ tạo hoặc do các nút trong add-on tạo, tên khoá khác nhau)
function rdNormalizeData(d) {
    const s = function(v) { return v == null ? '' : String(v); };
    const vocabulary = (Array.isArray(d.vocabulary) ? d.vocabulary : []).filter(function(v) { return v && (v.term || v.word); }).slice(0, RD_MAX_WORDS).map(function(v) {
        return {
            term: s(v.term || v.base_form || v.word), in_text: s(v.in_text || v.word), pos: s(v.pos),
            meaning_vi: s(v.meaning_vi), example: s(v.example || v.example_from_reading), example_vi: s(v.example_vi || v.translation)
        };
    });
    const structures = (Array.isArray(d.structures) ? d.structures : []).filter(function(x) { return x && (x.pattern || x.structure); }).slice(0, RD_MAX_STRUCTURES).map(function(x) {
        return {
            pattern: s(x.pattern || x.structure), in_text: s(x.in_text), explain_vi: s(x.explain_vi || x.explanation_vi),
            example: s(x.example || x.example_from_reading), example_vi: s(x.example_vi || x.translation)
        };
    });
    return { v: 1, at: d.at || Date.now(), vocabulary: vocabulary, structures: structures };
}
// Lấy bản mới hơn giữa field (đồng bộ từ máy tính) và bản lưu trong máy này.
// Field chứa HTML kiểu cũ thì giữ lại để xem trong mục gập.
function rdLoadData() {
    const st = rdState, fieldEl = spEl('rd-vocab-field');
    const raw = fieldEl ? (fieldEl.textContent || '').trim() : '';
    let field = null;
    if (raw) {
        try { const d = JSON.parse(raw); if (d && d.v === 1) field = d; } catch (e) { /* HTML kiểu cũ */ }
    }
    const best = [field, spStoreGet(st.storeKey)].filter(Boolean).sort(function(a, b) { return (b.at || 0) - (a.at || 0); })[0];
    if (best) return rdNormalizeData(best);
    if (raw) {
        const structEl = spEl('rd-struct-field');
        st.legacyHtml = fieldEl.innerHTML + (structEl ? structEl.innerHTML : '');
    }
    return null;
}
// Tô từ khoá và gạch chân các cụm từ vựng (bấm để xem nghĩa) trong bài đọc
function rdMarkText(text) {
    const st = rdState, lower = text.toLowerCase(), ranges = [];
    if (st.data) {
        st.data.vocabulary.forEach(function(v, i) {
            const needle = (v.in_text || v.term).trim().toLowerCase();
            const pos = needle.length > 1 ? lower.indexOf(needle) : -1;
            if (pos >= 0) ranges.push({ start: pos, end: pos + needle.length, kind: 'term', idx: i });
        });
    }
    const kw = st.word.trim();
    if (kw) {
        const esc = kw.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
        // Tiếng Anh: bắt cả dạng biến đổi (plan -> plans, planned)
        const re = /^[a-z]/i.test(kw) ? new RegExp('\\b' + esc + '\\w*', 'gi') : new RegExp(esc, 'g');
        let m;
        while ((m = re.exec(text))) ranges.push({ start: m.index, end: m.index + m[0].length, kind: 'key' });
    }
    ranges.sort(function(a, b) { return a.start - b.start || (b.end - b.start) - (a.end - a.start); });
    const kept = [];
    ranges.forEach(function(r) {
        const last = kept[kept.length - 1];
        if (last && r.start < last.end) {
            // Chồng lấn: giữ cụm từ vựng (bấm được); nếu cụm chứa từ khoá thì tô cả hai
            if (last.kind === 'term' && r.kind === 'key') last.hasKey = true;
            else if (last.kind === 'key' && r.kind === 'term') { r.hasKey = true; kept[kept.length - 1] = r; }
            return;
        }
        kept.push(r);
    });
    let html = '', cur = 0;
    kept.forEach(function(r) {
        if (r.start < cur) return;
        html += spEsc(text.slice(cur, r.start));
        const seg = spEsc(text.slice(r.start, r.end));
        html += r.kind === 'term'
            ? '<span class="rd-term tappable' + (r.hasKey ? ' rd-key' : '') + '" data-rd-i="' + r.idx + '">' + seg + '</span>'
            : '<mark class="rd-key">' + seg + '</mark>';
        cur = r.end;
    });
    return (html + spEsc(text.slice(cur))).replace(/\n/g, '<br>');
}

async function rdCreateVocab() {
    const st = rdState;
    if (!st || st.busy || !st.reading) return;
    const token = st.token;
    st.busy = true;
    rdRenderVocab();
    try {
        const d = rdNormalizeData(await geminiGenerateJson(rdBuildPrompt(), RD_VOCAB_SCHEMA, 0.3));
        if (!rdIsCurrent(token)) return;
        if (!d.vocabulary.length) throw new Error('AI không chọn được từ vựng nào, thử lại nhé');
        d.at = Date.now();
        st.data = d;
        st.legacyHtml = '';
        spStoreSet(st.storeKey, d);
        if (hasAnkiBridge()) {
            pycmd('rv_save:' + JSON.stringify({ word: st.word, data: d }), function(res) {
                if (res !== 'ok') console.warn('[Áp dụng] Không ghi được vào field Reading_Vocabulary:', res);
            });
        }
        rdRenderReading();   // gạch chân các cụm vừa tạo trong bài
    } catch (e) {
        if (rdIsCurrent(token)) { console.error('[Áp dụng]', e); showPracticeMessage('❌ ' + e.message, 'error'); }
    } finally {
        if (rdIsCurrent(token)) { st.busy = false; rdRenderVocab(); }
    }
}

    } else {

        try { localStorage.removeItem('userSpellingInput'); } catch (e) { /* private/restricted webview */ }
    }

    const correctWord = document.querySelector('.word')?.textContent?.trim() || '';

// Hàm so sánh chính tả giữ nguyên
function compareSpelling(correct,input){
    const r = {
        isPerfectMatch: false,
        isPartialMatch: false,
        correctWord: correct,
        userInput: input,
        charResults: []
    };
    
    if (input.toLowerCase() === correct.toLowerCase()){
        r.isPerfectMatch = true;
        r.charResults = input.split('').map(c => ({char: c, status: 'correct', position: 'exact'}));
        return r;
    }
    
    const maxLength = Math.max(correct.length, input.length);
    for(let i = 0; i < maxLength; i++){
        if(i >= input.length) break;
        const ch = input[i];
        let status = 'incorrect';
        
        if(i < correct.length){
            const cc = correct[i];
            if(ch.toLowerCase() === cc.toLowerCase()) status = 'correct';
            else if(correct.toLowerCase().includes(ch.toLowerCase())) status = 'partial';
        }
        r.charResults.push({char: ch, status});
    }
    
    r.isPartialMatch = r.charResults.some(c => c.status !== 'incorrect');
    return r;
}

// Hàm hiển thị kết quả giữ nguyên
function displaySpellingResult(result, correctWord, userInput){
    const res = document.getElementById('spellCheckResult');
    const hint = document.getElementById('spellCheckHint');

    if(result.isPerfectMatch){
        res.innerHTML = userInput.split('').map(c => `<span class="spell-check-char correct-char">${c}</span>`).join('');
        hint.innerHTML = '✅ Chính xác! Từ được viết đúng chính tả.';
        return;
    }
    
    if(!result.isPartialMatch){
        res.innerHTML = userInput.split('').map(c => `<span class="spell-check-char incorrect-char">${c}</span>`).join('');
        hint.innerHTML = `❌ Sai hoàn toàn. Từ đúng là: <strong>${correctWord}</strong>`;
        return;
    }
    
    let html = '';
    result.charResults.forEach(cr => {
        let cls = 'spell-check-char ';
        if(cr.status === 'correct') cls += 'correct-char';
        else if(cr.status === 'partial') cls += 'partial-correct';
        else cls += 'incorrect-char';
        html += `<span class="${cls}">${cr.char}</span>`;
    });
    res.innerHTML = html;

    const correctCount = result.charResults.filter(c => c.status === 'correct').length;
    if(userInput.length < correctWord.length) {
        hint.innerHTML = `⚠️ Thiếu ký tự. Từ đúng có ${correctWord.length} ký tự.`;
    } else if(userInput.length > correctWord.length) {
        hint.innerHTML = `⚠️ Thừa ký tự. Từ đúng có ${correctWord.length} ký tự.`;
    } else {
        hint.innerHTML = `🔍 Có ${correctCount}/${correctWord.length} ký tự đúng. Từ đúng: <strong>${correctWord}</strong>`;
    }
}

// Bộ thủ: lấy đúng chữ của bộ. AI đôi khi ghi cả chữ gốc (专) vào "character", còn bộ thật nằm trong ngoặc ở tên ("Bộ Thốn (寸)")
function radicalGlyph(r, mainWord) {
  const ch = String(r.character || '').trim();
  const HAN = /^[⺀-⿟㐀-鿿豈-﫿\ud840-𯰀-\udfff]+$/;
  const nm = String(r.name || '').trim();
  // "name": "厂" (chữ trơn) hoặc "Bộ Thốn (寸)" (trong ngoặc)
  const m = HAN.test(nm) ? [nm, nm] : (String(r.name_vi || '') + ' ' + nm).match(/[(（]\s*([⺀-⿟㐀-鿿豈-﫿\ud840-𯰀-\udfff]+)\s*[)）]/);
  const host = String(r.in_char || '').trim();
  if (ch && mainWord.indexOf(ch) >= 0 && m && m[1] !== ch) return { glyph: m[1], host: ch };
  const glyph = ch || (m ? m[1] : '');
  const chars = Array.from(mainWord);
  return { glyph: glyph, host: host || (chars.length === 1 && chars[0] !== glyph ? chars[0] : '') };
}
function renderRadicalsBody() {
  const dataEl = document.getElementById('radicals-data');
  const containerEl = document.getElementById('radicals-container');
  const mainWord = document.querySelector('.word')?.textContent?.trim() || '';
  
  if (!containerEl) return;

  if (!mainWord) {
    containerEl.innerHTML = `<div class="no-radicals-message"><div class="item-title">ℹ️ Thông tin</div><div class="item-content">Không có từ để phân tích.</div></div>`;
    return;
  }

  const raw = (dataEl?.textContent || '').trim();
  
  if (!raw || raw === '{}' || raw === '[]') {
    createDefaultRadicalAnalysis(mainWord, containerEl);
    return;
  }

  if (/^[\[{]/.test(raw)) {
    try {
      const data = JSON.parse(raw);
      renderRadicalsFromJSON(data, mainWord, containerEl);
    } catch (e) {
      console.warn('Radicals JSON parse failed, using default analysis:', e);
      createDefaultRadicalAnalysis(mainWord, containerEl);
    }
  } else {
    containerEl.innerHTML = `<div class="tab-item radicals-fallback">${raw}</div>`;
  }
}

function renderRadicalsFromJSON(data, mainWord, containerEl) {
  const arr = Array.isArray(data) ? data : [data];
  let finalHtml = '';

  // AI hiện trả về danh sách bộ thủ: {character, name_vi, name, meaning_vi, meaning, stroke_count, position, variant}
  // Dữ liệu kiểu cũ (phân tích từng chữ) có overall_explanation_vi/components -> vẫn hiển thị như trước
  const isRadicalItem = it => it && it.type !== 'memory_story' && !it.components && !it.overall_explanation_vi
    && (it.name_vi || it.meaning_vi || it.position || it.name || it.meaning);
  const radicalItems = arr.filter(isRadicalItem);

  if (radicalItems.length > 0) {
    finalHtml += `
      <div class="character-analysis">
        <div class="analysis-header">
          <h3 class="analysis-title">Bộ thủ trong chữ: <span class="analysis-char">${mainWord}</span></h3>
        </div>
        <div class="components-list">`;
    radicalItems.forEach(r => {
      const name = r.name_vi || r.name || '';
      const extraName = (r.name && r.name !== name) ? ` (${r.name})` : '';
      const strokes = parseInt(r.stroke_count, 10);
      const g = radicalGlyph(r, mainWord);
      finalHtml += `
          <div class="component-item" data-stroke-char="${spEsc(g.glyph)}" data-stroke-host="${spEsc(g.host)}" style="cursor:pointer" title="Bấm để xem cách viết">
            <div class="component-char">${spEsc(g.glyph)}</div>
            <div class="component-details">
              <div class="component-header">
                <span class="component-pinyin">${name}${extraName}</span>
                ${strokes > 0 ? `<span class="component-role other">${strokes} nét</span>` : ''}
              </div>
              <div class="component-meaning">${r.meaning_vi || r.meaning || ''}</div>
              ${r.position ? `<div class="component-meaning">📍 Vị trí: ${r.position}${g.host ? ' (trong chữ ' + spEsc(g.host) + ')' : ''}</div>` : ''}
              ${r.variant ? `<div class="component-meaning">🔁 Biến thể: ${r.variant}</div>` : ''}
              ${g.glyph ? `<div class="component-meaning"><b>▶️ Bấm để xem cách viết bộ này</b></div>` : ''}
            </div>
          </div>`;
    });
    finalHtml += `
        </div>
      </div>`;
  }

  arr.forEach((analysis, index) => {
    if (analysis?.type === 'memory_story' || isRadicalItem(analysis)) return;

    const ch = analysis?.character?.trim() || mainWord[index] || mainWord;
    if (!ch) return;

    const strokeCount = (analysis?.stroke_count || getStrokeCount(ch)).toString().trim();
    
    finalHtml += `
      <div class="character-analysis">
        <div class="analysis-header">
          <h3 class="analysis-title">Phân tích chữ: <span class="analysis-char">${ch}</span></h3>
          ${strokeCount ? `
            <div class="char-stroke-count" onclick="openFanningStrokesModal('${ch}')">
              ✍️ ${strokeCount} nét
            </div>
          ` : ''}
        </div>
        
        <div class="character-construction">
          <div class="construction-title">🏗️ Quá trình hình thành chữ:</div>
          <div class="stroke-breakdown">
            ${analysis?.overall_explanation_vi || generateStrokeBreakdown(ch)}
          </div>
        </div>
    `;

    if (Array.isArray(analysis?.components) && analysis.components.length > 0 && !analysis.is_pictograph) {
      finalHtml += `<div class="components-list">`;
      analysis.components.forEach(comp => {
        if (!comp) return;
        const role = comp.role || '';
        const roleClass = role.includes('Nghĩa') ? 'semantic' : (role.includes('Âm') ? 'phonetic' : 'other');
        finalHtml += `
          <div class="component-item" data-stroke-char="${spEsc(comp.component || '')}" data-stroke-host="${spEsc(ch)}" style="cursor:pointer" title="Bấm để xem cách viết">
            <div class="component-char">${comp.component || ''}</div>
            <div class="component-details">
              <div class="component-header">
                <span class="component-pinyin">${comp.pinyin || ''}</span>
                <span class="component-role ${roleClass}">${role}</span>
              </div>
              <div class="component-meaning">${comp.meaning_vi || comp.meaning || ''}</div>
            </div>
          </div>
        `;
      });
      finalHtml += `</div>`;
    }

    finalHtml += `</div>`;
  });

  if (finalHtml) {
    containerEl.innerHTML = finalHtml;
  } else {
    // Chỉ có câu chuyện ghi nhớ (hoặc không có gì) -> phân tích mặc định, không gán kết quả undefined
    createDefaultRadicalAnalysis(mainWord, containerEl);
  }
}

// Hàm trích xuất Hán tự và Pinyin
function extractHanziAndPinyin(text) {
    let hanzi = text, pinyin = '';
    const matches = text.match(/\(([^)]+)\)/g);
    if (matches) {
        hanzi = text.replace(/\([^)]*\)/g, '').replace(/\s+/g, ' ').trim();
        pinyin = matches.map(m => m.slice(1, -1)).join(' ').replace(/\s+/g, ' ').trim();
    }
    return {hanzi, pinyin};
}

function processChineseExample(){
  const exEl=document.getElementById('example-en-data'); if(!exEl) return;
  const exampleEn=exEl.textContent;
  const mainWord=document.querySelector('.word')?.textContent||'';
  const isChinese=/[\u4e00-\u9fff]/.test(mainWord);
  const hanziElement=document.getElementById('example-hanzi');
  const pinyinElement=document.getElementById('example-pinyin');
  const ttsBtn=document.getElementById('example-tts-btn');

  if(hanziElement && pinyinElement && ttsBtn){
    if(isChinese && exampleEn){
      const {hanzi,pinyin}=extractHanziAndPinyin(exampleEn);
      hanziElement.textContent=hanzi;
      if(pinyin){ pinyinElement.textContent=pinyin; pinyinElement.style.display='block'; }
      else pinyinElement.style.display='none';
      ttsBtn.onclick=function(event){ 
          event.stopPropagation();
          speakChineseText(event, exampleEn); 
      };
    }else{
      hanziElement.textContent=exampleEn;
      pinyinElement.style.display='none';
      ttsBtn.onclick=function(event){ 
          event.stopPropagation();
          speakTextWithEdge(exampleEn, event); 
      };
    }
  }
}

// Tab không có dữ liệu (xem #tab-has-data) thì ẩn cả nút lẫn nội dung. Luyện câu luôn hiện (AI trong thẻ tự ra đề).
// Thẻ không có ví dụ, bài đọc, mở rộng = thẻ thêm nhanh -> hiện banner "✨ Thêm chi tiết".
function liteSetupTabs() {
    var hidden = {};
    document.querySelectorAll('#tab-has-data [data-tab]').forEach(function(mark) {
        if (!mark.textContent.trim()) hidden[mark.getAttribute('data-tab')] = true;
    });
    document.querySelectorAll('.tabs .tab-link').forEach(function(btn) {
        var m = (btn.getAttribute('onclick') || '').match(/openTab\(event,\s*'(\w+)'\)/);
        if (!m || !hidden[m[1]]) return;
        btn.style.display = 'none';
        var content = document.getElementById(m[1]);
        if (content) content.style.display = 'none';
    });
    if (hidden.ViDu && hidden.MoRong && hidden.ApDung) {
        document.getElementById('lite-banner').style.display = 'block';
    }
}

function liteFillFull() {
    var btn = document.getElementById('lite-fill-btn');
    var note = document.getElementById('lite-note');
    if (!hasAnkiBridge()) {
        // AnkiMobile / AnkiDroid không chạy add-on -> điền trên máy tính rồi đồng bộ về

        return;

    var word = (document.querySelector('.word') || {}).textContent || '';
    btn.disabled = true;

        btn.textContent = '✨ Thêm chi tiết';

    });
}
