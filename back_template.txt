    }

function getEdgeVoice(text) {
    const lang = detectLanguage(text);
    switch (lang) {
        case 'zh': return 'zh-CN-XiaoxiaoNeural';
        case 'ja': return 'ja-JP-NanamiNeural';
        case 'ko': return 'ko-KR-SunHiNeural';
        case 'vi': return 'vi-VN-NamMinhNeural';
        case 'de': return 'de-DE-KatjaNeural';
        default:   return 'en-GB-RyanNeural'; // Force UK voice
    }
}

async function speakWithEdgeReal(text, event) {
    if (event) event.stopPropagation();
    
    const cleanText = text ? text.replace(/[\(\[\{\<][^\)\]\}\>]*[\)\]\}\>]/g, '').trim() : '';
    if (!cleanText) return;

    let btn = event?.target;
    if (btn && btn.tagName !== 'BUTTON') btn = btn.closest('button');
    
    // --- Cập nhật UI ---
    if (btn) {
        document.querySelectorAll('.tts-btn').forEach(b => {
            b.classList.remove('playing');
            if (b.classList.contains('gemini-btn')) b.innerHTML = '🔊';
            if (b.classList.contains('edge-btn')) b.innerHTML = '⚡';
        });
        
        btn.classList.add('playing');
        btn.innerHTML = '⏹️'; // Icon Stop khi đang đọc (theo code mẫu)
    }

    // --- Dừng các luồng âm thanh cũ ---
    if (window.speechSynthesis) window.speechSynthesis.cancel();
    if (typeof audioCtx !== 'undefined' && audioCtx && audioCtx.state === 'running') {
        try { audioCtx.close(); } catch(e){}
    }

    if (!window.speechSynthesis) return;

    // --- Web Speech API Logic (Thay thế Fetch Edge API để chạy được trên Mobile) ---
    const u = new SpeechSynthesisUtterance(cleanText);
    
    // Phát hiện ngôn ngữ (dựa theo logic code mẫu)
    const isChinese = /[\u4e00-\u9fff\u3400-\u4dbf]/.test(cleanText);
    const lang = isChinese ? 'zh' : 'en';

    // Cấu hình giọng đọc
    if(lang === 'zh'){
        u.lang = 'zh-CN';
        u.rate = 0.8;
        u.pitch = 1.0;
    } else {
        u.lang = 'en-GB'; // Force UK accent
        u.rate = 0.9;
        u.pitch = 1.0;
    }

    // Hàm phát âm thanh
    const play = () => {
        const voices = window.speechSynthesis.getVoices();
        let v = null;
        if(lang === 'zh') {
            v = voices.find(x => x.lang.includes('zh') || x.lang.includes('cn') || x.name.toLowerCase().includes('chinese'));
        } else {
            // Ưu tiên giọng UK (en-GB), fallback sang bất kỳ giọng en nào
            v = voices.find(x => x.lang === 'en-GB')
              || voices.find(x => x.lang.startsWith('en-GB'))
              || voices.find(x => x.name.toLowerCase().includes('british'))
              || voices.find(x => x.lang.startsWith('en'));
        }
        
        if(v) u.voice = v;
        window.speechSynthesis.speak(u);
    };

    // Sự kiện khi đọc xong hoặc lỗi
    const reset = () => {
        if (btn) {
            btn.classList.remove('playing');
            btn.innerHTML = '⚡'; // Trả về icon tia sét ban đầu
        }
    };
    u.onend = reset;
    u.onerror = reset;

    // Xử lý trường hợp danh sách giọng đọc chưa tải xong
    if (window.speechSynthesis.getVoices().length === 0) {
        window.speechSynthesis.onvoiceschanged = play;
    } else {
        play();
    }
}
function speakExampleEdge(event) {
    const exEl = document.getElementById('example-en-data');
    if (!exEl) return;
    
    let text = exEl.textContent;
    const mainWord = document.querySelector('.word')?.textContent || '';

        text = text.replace(/\([^)]*\)/g, '').replace(/\s+/g, ' ').trim();
    }
    
    speakWithEdgeReal(text, event);
}

    // Reset mọi trạng thái cũ trước khi bắt đầu mới
    cleanupOldRequests();
    
    // Gọi hàm chính
    speakWithGemini(text, event);
}

// Hàm dừng TTS
function stopTTS() {
    if (audioCtx && audioCtx.state === 'running') {
        audioCtx.close();
        audioCtx = null;
    }
    if (window.speechSynthesis) {
        window.speechSynthesis.cancel();
    }
    resetTTSIcons();
}

// Hàm dọn dẹp các request cũ đang treo
function cleanupOldRequests() {
    if (globalTimeoutId) {
        clearTimeout(globalTimeoutId);
        globalTimeoutId = null;
    }
    if (globalTTSAbortController) {
        globalTTSAbortController.abort(); // Ngắt kết nối mạng ngay lập tức
        globalTTSAbortController = null;
    }
    // Dừng Web Speech cũ nếu đang nói
    if (window.speechSynthesis) window.speechSynthesis.cancel();
}

async function speakWithGemini(text, event, keyIndex = 0) {
    if (!text || !text.trim()) return;

    // ========================================================================
    // [NEW] LOGIC ÉP BUỘC ĐỌC THEO IPA CHÍNH XÁC
    // Kiểm tra xem đoạn text đang cần đọc có phải là từ vựng chính không.
    // Nếu đúng, chèn IPA vào prompt để ép Gemini-2.5-pro-audio đọc chuẩn xác.
    // ========================================================================
    let promptForAudio = text;
    const mainWord = document.querySelector('.word')?.textContent?.trim() || '';
    
    if (text.trim() === mainWord) {
        const currentIpa = document.querySelector('.ipa')?.textContent?.trim() || '';
        if (currentIpa && !currentIpa.includes('{' + '{')) {
            // Director-style instruction for TTS: ép giọng UK với IPA
            promptForAudio = `Say the word '${text}' in a British RP accent, matching this pronunciation: /${currentIpa}/.`;
        } else {
            promptForAudio = `Say the word '${text}' in a British RP accent.`;
        }
    } else {
        const isEnglish = detectLanguage(text) === 'en';
        if (isEnglish) {
            promptForAudio = `Read this in a British RP accent: ${text}`;
        }
    }
    // ========================================================================
    
    // Lần thử đầu lấy danh sách key mới; các lần thử key tiếp theo dùng lại đúng danh sách đó
    if (keyIndex === 0 || !ttsKeySnapshot) ttsKeySnapshot = getGeminiTTSKeys();
    var keys = ttsKeySnapshot;

    // Nếu không có key, fallback ngay
    if (keys.length === 0) {
        speakWithWebSpeechFallback(text, event);
        return;
    }

    var currentKey = keys[keyIndex % keys.length];

    // --- XỬ LÝ GIAO DIỆN (Chỉ lần đầu) ---
    if (keyIndex === 0) {
        if (audioCtx && audioCtx.state === 'running') {
            try { audioCtx.close(); } catch(e){}
            audioCtx = null;
        }
        
        document.querySelectorAll('.tts-btn, .tts-btn-small').forEach(function(b) {
            b.classList.remove('playing');
            if (b.classList.contains('gemini-btn')) b.innerHTML = '🔊';
            if (b.classList.contains('edge-btn')) b.innerHTML = '⚡';
            if (!b.classList.contains('gemini-btn') && !b.classList.contains('edge-btn')) b.innerHTML = '🔊';
        });
        
        var button = event?.target || document.querySelector('.tts-btn, .tts-btn-small');
        if (button && button.tagName !== 'BUTTON') button = button.closest('button');
        currentGeminiTTSButton = button;
        
        if (button) {
            button.classList.add('playing');
            button.innerHTML = '⏳';
        }
    } else {
        var button = currentGeminiTTSButton;
    }
    
    // --- CƠ CHẾ TIMEOUT CHẮC CHẮN CHO MOBILE ---
    let isRequestFinished = false; // Cờ kiểm soát: true = đã xong (thành công hoặc lỗi)
    const controller = new AbortController();
    
    // Thiết lập bộ đếm giờ 5 giây
    const timeoutId = setTimeout(() => {
        if (isRequestFinished) return; // Nếu API đã trả về thì thôi
        
        console.warn("⚠️ Hết giờ (5s)! Bắt đầu chuyển sang giọng đọc máy...");
        isRequestFinished = true; // Đánh dấu là xong để chặn API nếu nó trả về sau đó
        controller.abort();       // Ngắt mạng
        
        // Chuyển ngay sang Fallback
        resetTTSIcons(); 
        speakWithWebSpeechFallback(text, event);
    }, 5000); 

    try {
        console.log(`Đang gọi Gemini TTS (Key #${keyIndex + 1})...`);
        var url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${currentKey}`;
        
        // Xác định languageCode dựa trên ngôn ngữ: en-GB cho tiếng Anh, 
        // các ngôn ngữ khác dùng locale tương ứng
        var detectedLang = detectLanguage(text);
        var ttsLangCode = 'en-GB'; // Mặc định UK English
        if (detectedLang === 'zh') ttsLangCode = 'zh-CN';
        else if (detectedLang === 'ja') ttsLangCode = 'ja-JP';
        else if (detectedLang === 'vi') ttsLangCode = 'vi-VN';
        else if (detectedLang === 'ko') ttsLangCode = 'ko-KR';
        
        var requestBody = {
            contents: [{parts: [{text: promptForAudio}]}],
            generationConfig: {
                responseModalities: ["AUDIO"],
                speechConfig: {
                    voiceConfig: { prebuiltVoiceConfig: { voiceName: GEMINI_TTS_VOICE } },
                    languageCode: ttsLangCode
                }
            }
        };
        
        console.log(`🇬🇧 TTS Config: lang=${ttsLangCode}, voice=${GEMINI_TTS_VOICE}`);
        
        var response = await fetch(url, {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify(requestBody),
            signal: controller.signal
        });

        // Nếu code chạy đến đây tức là chưa bị timeout ngắt
        if (isRequestFinished) return; // Nếu timeout đã chạy trước đó thì bỏ qua kết quả này
        clearTimeout(timeoutId);       // Hủy bộ đếm giờ vì đã có kết quả

        if (!response.ok) {
            // Key bị xoá/thu hồi -> ghi nhớ để lần sau bỏ qua (dùng chung với Luyện câu, Áp dụng)
            if (response.status === 400) {
                const err = await geminiReadError(response);
                if (/API_KEY/.test(err.reason + err.message)) geminiMark('dead', geminiKeyId(currentKey), 24 * 3600 * 1000);
            }
            // Nếu lỗi server/limit và còn key -> Thử key tiếp theo
            if ([429, 500, 503, 403].includes(response.status)) {
                if (keyIndex < keys.length - 1) {
                    await speakWithGemini(text, event, keyIndex + 1);
                    return;
                }
            }
            throw new Error(`Lỗi API: ${response.status}`);
        }

        var data = await response.json();
        
        if (data.candidates && data.candidates[0].content && data.candidates[0].content.parts[0].inlineData) {
            var audioBase64 = data.candidates[0].content.parts[0].inlineData.data;
            
            // Đánh dấu đã xong để timeout không chạy nữa
            isRequestFinished = true; 
            await playPcmAudio(audioBase64, button, text);
        } else {
            throw new Error("Không có audio data");
        }

    } catch (e) {
        if (isRequestFinished) return; // Nếu timeout đã xử lý rồi thì thôi
        
        isRequestFinished = true;
        clearTimeout(timeoutId);
        
        console.error("Gemini TTS Error:", e);
        
        // Nếu lỗi mạng (không phải do timeout ngắt) và còn key -> Thử tiếp
        if (e.name !== 'AbortError' && keyIndex < keys.length - 1) {
             await speakWithGemini(text, event, keyIndex + 1);
             return;
        }
        
        // Fallback cuối cùng
        resetTTSIcons();
        speakWithWebSpeechFallback(text, event);
    }
}

// Hàm giải mã và phát âm thanh PCM
async function playPcmAudio(base64String, button, originalText) {
    try {
        var binaryString = atob(base64String);
        var len = binaryString.length;
        var bytes = new Uint8Array(len);
        for (var i = 0; i < len; i++) {
            bytes[i] = binaryString.charCodeAt(i);
        }
        
        var int16Data = new Int16Array(bytes.buffer);
        var float32Data = new Float32Array(int16Data.length);
        for (var i = 0; i < int16Data.length; i++) {
            float32Data[i] = int16Data[i] / 32768.0;
        }

        if (!audioCtx || audioCtx.state === 'closed') {
            audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        }
        
        // QUAN TRỌNG: Resume context cho Mobile
        if (audioCtx.state === 'suspended') {
            await audioCtx.resume();
        }
        
        var buffer = audioCtx.createBuffer(1, float32Data.length, 24000);
        buffer.copyToChannel(float32Data, 0);
        
        var source = audioCtx.createBufferSource();
        source.buffer = buffer;
        source.connect(audioCtx.destination);
        
        source.onended = function() {
            resetTTSIcons();
        };
        
        if (button) {
            if (button.classList.contains('gemini-btn')) {
                button.innerHTML = '🔊';
            } else if (button.classList.contains('edge-btn')) {
                button.innerHTML = '⚡';
            } else {
                button.innerHTML = '🔊';
            }
        }
        
        source.start(0);
        
    } catch (e) {
        console.error("Audio Playback Error:", e);
        resetTTSIcons();
        speakWithWebSpeechFallback(originalText, null);
    }
}

function resetTTSIcons() {
    document.querySelectorAll('.tts-btn, .tts-btn-small').forEach(function(b) {
        b.classList.remove('playing');
        if (b.classList.contains('gemini-btn')) b.innerHTML = '🔊';
        else if (b.classList.contains('edge-btn')) b.innerHTML = '⚡';
        else b.innerHTML = '🔊';
    });
    currentGeminiTTSButton = null;
}

// Fallback: Web Speech API (Cực kỳ đơn giản và nhanh)
function speakWithWebSpeechFallback(text, event) {
    if (!window.speechSynthesis) return;
    
    console.log(">> Running Fallback TTS");
    
    // Đảm bảo dừng mọi thứ trước đó
    window.speechSynthesis.cancel();
    
    const u = new SpeechSynthesisUtterance(text);
    
    // Cố gắng chọn giọng tốt nhất
    const lang = detectLanguage(text);
    if (lang === 'zh') u.lang = 'zh-CN';
    else if (lang === 'vi') u.lang = 'vi-VN';
    else if (lang === 'ja') u.lang = 'ja-JP';
    else {
        u.lang = 'en-GB'; // Force UK accent
        // Tìm giọng UK trong danh sách
        const voices = window.speechSynthesis.getVoices();
        const ukVoice = voices.find(x => x.lang === 'en-GB')
                     || voices.find(x => x.lang.startsWith('en-GB'))
                     || voices.find(x => x.name.toLowerCase().includes('british'));
        if (ukVoice) u.voice = ukVoice;
    }
    
    u.rate = 0.9;
    
    // Hiệu ứng nút
    var btn = currentGeminiTTSButton;
    // Nếu không tìm thấy biến toàn cục, tìm lại qua event
    if (!btn && event && event.target) {
        btn = event.target.closest('button') || event.target;
    }

    if (btn) {
        btn.classList.add('playing');
        btn.innerHTML = '🗣️'; // Icon khác để biết là fallback
        
        u.onstart = function() { 
            if(btn) btn.innerHTML = '⏹️'; 
        };
        u.onend = function() { 
            if(btn) {
                btn.classList.remove('playing');
                if (btn.classList.contains('gemini-btn')) btn.innerHTML = '🔊';
                else if (btn.classList.contains('edge-btn')) btn.innerHTML = '⚡';
                else btn.innerHTML = '🔊';
            }
        };
        u.onerror = function() {
            if(btn) {
                btn.classList.remove('playing');
                if (btn.classList.contains('gemini-btn')) btn.innerHTML = '🔊';
                else if (btn.classList.contains('edge-btn')) btn.innerHTML = '⚡';
                else btn.innerHTML = '🔊';
            }
        };
    }
    
    window.speechSynthesis.speak(u);
}

function detectLanguage(text) {
    if (!text) return 'en';
    if (/[\u4E00-\u9FFF]/.test(text)) return 'zh';
    if (/[àáạảãâầấậẩẫăằắặẳẵèéẹẻẽêềếệểễìíịỉĩòóọỏõôồốộổỗơờớợởỡùúụủũưừứựửữỳýỵỷỹđ]/.test(text.toLowerCase())) return 'vi';
    if (/[\u3040-\u309f\u30a0-\u30ff]/.test(text)) return 'ja';
    return 'en';
}

// Hàm xử lý đọc văn bản tiếng Trung (CẬP NHẬT)
function speakChineseText(event, text) {
    if (event) event.stopPropagation();
    
    // THÊM DÒNG NÀY: Ngắt các luồng âm thanh cũ để tránh lỗi trên Mobile/chồng âm
    if (typeof cleanupOldRequests === 'function') cleanupOldRequests();
    
    var cleanText = text;
    if (text) {
        // Xóa nội dung trong mọi loại ngoặc
        cleanText = cleanText.replace(/[\(\[\{\<][^\)\]\}\>]*[\)\]\}\>]/g, '').trim();
    }
    
    if (!cleanText) cleanText = text;
    
    speakWithGemini(cleanText, event);
}


/* ==================== HIỂN THỊ CONTEXT TRONG TAB BO SUNG ==================== */
function displayContextInBoSung() {
    const contextElement = document.querySelector('.context-content');
    const languageIndicator = document.getElementById('context-language-indicator');
    
    if (!contextElement || !languageIndicator) return;
    
    const contextText = contextElement.textContent || contextElement.innerText;
    if (!contextText.trim()) return;
    
    const detectedLang = detectContextLanguage(contextText);
    const contextCategory = categorizeContext(contextText);
    
    // Chỉ hiển thị indicator cho 4 ngôn ngữ được hỗ trợ
    if (detectedLang && detectedLang !== "Unknown") {
        let indicatorText = `🌐 Ngôn ngữ context: ${detectedLang}`;
        if (contextCategory !== 'general') {
            indicatorText += ` | 📂 Chủ đề: ${getContextCategoryName(contextCategory)}`;
        }
        languageIndicator.innerHTML = `<small>${indicatorText}</small>`;
        languageIndicator.style.display = 'block';
    } else {
        languageIndicator.style.display = 'none';
    }
}

// Hàm phát hiện ngôn ngữ context - CHỈ 4 NGÔN NGỮ
function detectContextLanguage(text) {
    if (!text) return "Unknown";
    
    // Japanese
    if (/[\u3040-\u30ff\u31f0-\u31ff]/.test(text)) return "Japanese";
    
    // Chinese
    if (/[\u4e00-\u9fff]/.test(text)) return "Chinese";
    
    // Vietnamese (các ký tự có dấu đặc trưng)
    if (/[àáạảãâầấậẩẫăằắặẳẵèéẹẻẽêềếệểễìíịỉĩòóọỏõôồốộổỗơờớợởỡùúụủũưừứựửữỳýỵỷỹđ]/i.test(text)) 
        return "Vietnamese";
    
    // Mặc định là English
    return "English";
}

// Hàm phân loại context trong JavaScript
function categorizeContext(text) {
    const textLower = text.toLowerCase();
    const categories = {
        'medical': ['y học', 'bệnh viện', 'bác sĩ', 'sức khỏe', 'y tế'],
        'technology': ['công nghệ', 'máy tính', 'phần mềm', 'lập trình'],
        'business': ['kinh doanh', 'thương mại', 'công ty', 'doanh nghiệp'],
        'academic': ['học thuật', 'trường học', 'giáo dục', 'nghiên cứu'],
        'daily': ['hàng ngày', 'giao tiếp', 'cuộc sống']
    };
    
    for (const [category, keywords] of Object.entries(categories)) {
        if (keywords.some(keyword => textLower.includes(keyword))) {
            return category;
        }
    }
    return 'general';
}

function getContextCategoryName(category) {
    const names = {
        'medical': 'Y học',
        'technology': 'Công nghệ',
        'business': 'Kinh doanh',
        'academic': 'Học thuật', 
        'daily': 'Hàng ngày',
        'general': 'Chung'
    };
    return names[category] || category;
}

/* ==================== XỬ LÝ 3 VÍ DỤ GỐC TỪ ==================== */
function renderEtymologyExamples() {
    const etymologyDataEl = document.getElementById('etymology-data');
    const examplesContainer = document.getElementById('etymology-examples-container');
    const examplesList = document.getElementById('etymology-examples-list');
    
    if (!etymologyDataEl || !examplesContainer) return;
    
    const rawData = etymologyDataEl.textContent.trim();
    
    if (!rawData || rawData === '{}' || rawData === '[]') {
        examplesContainer.style.display = 'none';
        return;
    }
    
    try {
        const etymologyData = JSON.parse(rawData);
        
        if (etymologyData.examples && Array.isArray(etymologyData.examples) && etymologyData.examples.length > 0) {
            let examplesHtml = '';
            
            etymologyData.examples.forEach((example, index) => {
                if (example && example.word) {
                    examplesHtml += `
                        <div class="etymology-example-item">
                            <div class="etymology-example-word">${example.word}</div>
                            ${example.explanation ? `<div class="etymology-example-explanation">${example.explanation}</div>` : ''}
                            ${example.sentence ? `<div class="etymology-example-sentence">${example.sentence}</div>` : ''}
                            ${example.translation ? `<div class="etymology-example-translation">${example.translation}</div>` : ''}
                        </div>
                    `;
                }
            });
            
            if (examplesHtml) {
                examplesList.innerHTML = examplesHtml;
                examplesContainer.style.display = 'block';
            } else {
                examplesContainer.style.display = 'none';
            }
        } else {
            examplesContainer.style.display = 'none';
        }
    } catch (error) {
        console.error('Lỗi phân tích dữ liệu etymology examples:', error);
        examplesContainer.style.display = 'none';
    }
}

/* ==================== OTHER MEANINGS ==================== */
function renderOtherMeanings() {
    const dataEl = document.getElementById('other-meaning-data');
    const containerEl = document.getElementById('other-meaning-container');
    const listEl = document.getElementById('other-meaning-list');
    
    if (!(dataEl && containerEl && listEl)) return;
    
    const rawData = dataEl.textContent.trim();
    
    if (!rawData || rawData === '{}' || rawData === '[]') {
        containerEl.style.display = 'none';
        return;
    }
    
    try {
        const otherMeanings = JSON.parse(rawData);
        
        if (Array.isArray(otherMeanings) && otherMeanings.length > 0) {
            let html = '';
            
            otherMeanings.forEach((meaning, index) => {
                if (meaning && meaning.meaning && meaning.meaning.trim() !== "") {
                    html += `
                        <div class="other-meaning-item">
                            <div class="other-meaning-header">
                                <span class="other-meaning-index">${index + 1}.</span>
                                <span class="other-meaning-text">${meaning.meaning}</span>
                            </div>
                            ${meaning.context_vi && meaning.context_vi.trim() !== "" ? `
                                <div class="other-meaning-context">
                                    <strong>📝 Ngữ cảnh sử dụng:</strong> ${meaning.context_vi}
                                </div>
                            ` : ''}
                            ${meaning.example_en && meaning.example_en.trim() !== "" ? `
                                <div class="other-meaning-example">
                                    <strong>💡 Ví dụ:</strong> ${meaning.example_en}
                                </div>
                            ` : ''}
                            ${meaning.example_vi && meaning.example_vi.trim() !== "" ? `
                                <div class="other-meaning-translation">
                                    <strong>🌐 Dịch:</strong> ${meaning.example_vi}
                                </div>
                            ` : ''}
                            ${meaning.usage_note_vi && meaning.usage_note_vi.trim() !== "" ? `
                                <div class="other-meaning-usage">
                                    <strong>🎯 Ghi chú sử dụng:</strong> ${meaning.usage_note_vi}
                                </div>
                            ` : ''}
                            ${meaning.sources && Array.isArray(meaning.sources) && meaning.sources.length > 0 ? `
                                <div class="other-meaning-sources">
                                    <strong>📚 Nguồn tham khảo:</strong> ${meaning.sources.join(', ')}
                                </div>
                            ` : ''}
                        </div>
                    `;
                }
            });
            
            if (html) {
                listEl.innerHTML = html;
                containerEl.style.display = 'block';
                
                // Hiển thị số lượng nghĩa chính xác
                const actualCount = otherMeanings.filter(m => m && m.meaning && m.meaning.trim() !== "").length;
                const titleEl = containerEl.querySelector('.group-title');
                if (titleEl) {
                    if (actualCount === 10) {
                        titleEl.innerHTML = `🔤 10 Nghĩa chính của từ`;
                    } else {
                        titleEl.innerHTML = `🔤 ${actualCount} Nghĩa chính của từ`;
                    }
                }
            } else {
                containerEl.style.display = 'none';
            }
        } else {
            containerEl.style.display = 'none';
        }
    } catch (error) {
        console.error('Lỗi phân tích dữ liệu other meanings:', error);
        containerEl.style.display = 'none';
    }
}

// Dùng cho luyện câu, tạo từ vựng, bài tập dịch (Python cũng gọi hàm này qua eval)
function showPracticeMessage(message, type) {
    type = type || 'info';
    let toast = document.getElementById('practice-toast');
    if (!toast) {
        toast = document.createElement('div');
        toast.id = 'practice-toast';
        toast.onclick = function() { toast.style.display = 'none'; };
        document.body.appendChild(toast);
    }
    toast.className = 'practice-toast toast-' + type;
    toast.textContent = message;
    toast.style.display = 'block';
    clearTimeout(window.__practiceToastTimer);
    if (type !== 'loading') {
        window.__practiceToastTimer = setTimeout(function() { toast.style.display = 'none'; }, type === 'error' ? 7000 : 3500);
    }

/* ==================== GỌI GEMINI DÙNG CHUNG (LUYỆN CÂU, ÁP DỤNG) ==================== */
// Gọi thẳng từ thẻ nên chạy được cả Anki máy tính lẫn điện thoại (dùng danh sách key GEMINI_TTS_API_KEYS ở trên).
// Thử lần lượt từng model: lượt miễn phí của Google tính riêng cho từng model trên mỗi project,
// nên có model dự phòng thì tổng số lượt dùng được nhiều hơn.
// Mỗi đời model tắt "suy nghĩ" theo cách khác nhau: 2.5 dùng thinkingBudget, 3.x dùng thinkingLevel.
var GEMINI_TEXT_MODELS = [
    { model: 'gemini-3.8-flash', thinking: { thinkingLevel: 'low' } },
    { model: 'gemini-2.5-flash', thinking: { thinkingBudget: 0 } },
    { model: 'gemini-3.5-flash-lite', thinking: { thinkingLevel: 'low' } },
    { model: 'gemini-flash-lite-latest', thinking: { thinkingLevel: 'low' } }
];
var GEMINI_TEXT_TIMEOUT_MS = 25000;
var GEMINI_MAX_CALLS = 8;           // tối đa ngần này lần gọi (qua các key/model) cho mỗi yêu cầu
var GEMINI_HEALTH_KEY = 'gemini_health_v1';
var geminiKeyStart = 0;

// Ghi nhớ trong máy: key chết (dead), key hết lượt với từng model (block), model đang quá tải (busy)
// -> lần sau bỏ qua ngay thay vì thử lại từng cái.
function geminiHealth() {
    const h = spStoreGet(GEMINI_HEALTH_KEY) || {};
    return { dead: h.dead || {}, block: h.block || {}, busy: h.busy || {} };
}
function geminiKeyId(key) { return key.slice(-8); }
function geminiMark(kind, id, ms) {
    const h = geminiHealth(), now = Date.now();
    h[kind][id] = now + ms;
    ['dead', 'block', 'busy'].forEach(function(k) {
        Object.keys(h[k]).forEach(function(x) { if (h[k][x] < now) delete h[k][x]; });
    });
    spStoreSet(GEMINI_HEALTH_KEY, h);
}
function geminiIsDeadKey(key) { return (geminiHealth().dead[geminiKeyId(key)] || 0) > Date.now(); }
async function geminiReadError(res) {
    const j = await res.json().catch(function() { return {}; });
    const e = j.error || {}, details = e.details || [];
    const reasons = details.map(function(d) { return d.reason; }).filter(Boolean);
    const quota = [];
    details.forEach(function(d) { (d.violations || []).forEach(function(v) { quota.push(v.quotaId || ''); }); });
    return { status: res.status, reason: reasons[0] || e.status || '', quota: quota.join(' '), message: e.message || '' };
}
function geminiFailMessage(stats, calls) {
    if (!calls) return 'Tất cả key/model đang tạm nghỉ (hết lượt hoặc quá tải). Thử lại sau ít phút.';
    if (stats.quota && !stats.busy && !stats.other.length) {
        return 'Đã hết lượt Gemini miễn phí trên các key còn dùng được. Thử lại sau, hoặc thêm key mới tạo ở tài khoản Google khác.';
    }
    const parts = [];
    if (stats.invalid) parts.push(stats.invalid + ' key không còn hợp lệ');
    if (stats.quota) parts.push(stats.quota + ' lần hết lượt');
    if (stats.busy) parts.push('model đang quá tải');
    return 'Không gọi được Gemini (' + parts.concat(stats.other.slice(-2)).join('; ') + ')';
}
async function geminiGenerateJson(prompt, schema, temperature) {
    const keys = getGeminiTTSKeys();
    if (!keys.length) throw new Error('Chưa có API key trong thẻ (GEMINI_TTS_API_KEYS)');
    const MINUTE = 60000, HOUR = 60 * MINUTE, DAY = 24 * HOUR;
    const stats = { invalid: 0, quota: 0, busy: 0, other: [] };
    const started = Date.now();
    let calls = 0;
    outer:
    for (const m of GEMINI_TEXT_MODELS) {
        if ((geminiHealth().busy[m.model] || 0) > Date.now()) continue;
        const generationConfig = { responseMimeType: 'application/json', responseSchema: schema, temperature: temperature };
        if (m.thinking) generationConfig.thinkingConfig = m.thinking;
        const body = JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], generationConfig: generationConfig });
        for (let n = 0; n < keys.length; n++) {
            const idx = (geminiKeyStart + n) % keys.length, key = keys[idx], kid = geminiKeyId(key);
            const h = geminiHealth();
            if ((h.dead[kid] || 0) > Date.now() || (h.block[m.model + '|' + kid] || 0) > Date.now()) continue;
            if (++calls > GEMINI_MAX_CALLS || Date.now() - started > 45000) break outer;
            const controller = new AbortController();
            const timer = setTimeout(function() { controller.abort(); }, GEMINI_TEXT_TIMEOUT_MS);
            try {
                const res = await fetch('https://generativelanguage.googleapis.com/v1beta/models/' + m.model + ':generateContent?key=' + encodeURIComponent(key), {
                    method: 'POST', headers: { 'Content-Type': 'application/json' }, body: body, signal: controller.signal
                });
                if (res.ok) {
                    const data = await res.json();
                    const parts = (data.candidates && data.candidates[0] && data.candidates[0].content && data.candidates[0].content.parts) || [];
                    const text = parts.filter(function(p) { return !p.thought; }).map(function(p) { return p.text || ''; }).join('');
                    if (!text) { stats.other.push(m.model + ': phản hồi trống'); continue; }
                    geminiKeyStart = idx;
                    console.log('[Gemini] ' + m.model + ' · key ' + (idx + 1) + ' · ' + (Date.now() - started) + 'ms');
                    return spParseJson(text);
                }
                const err = await geminiReadError(res);
                console.warn('[Gemini] ' + m.model + ' · key ' + (idx + 1) + ' lỗi ' + err.status + ' ' + err.reason + ': ' + err.message.slice(0, 200));
                if (err.status === 400 && /API_KEY/.test(err.reason + err.message)) {
                    stats.invalid++;
                    geminiMark('dead', kid, DAY);                         // key bị xoá/thu hồi
                } else if (err.status === 429) {
                    stats.quota++;
                    geminiMark('block', m.model + '|' + kid, /PerDay/i.test(err.quota) ? 3 * HOUR : MINUTE);
                } else if (err.status === 404 || err.status === 403) {
                    stats.other.push(m.model + ': không dùng được với key ' + (idx + 1));
                    geminiMark('block', m.model + '|' + kid, err.status === 404 ? 7 * DAY : DAY);
                } else if (err.status >= 500) {
                    stats.busy++;
                    geminiMark('busy', m.model, 10 * MINUTE);             // model quá tải: sang model khác
                    continue outer;
                } else {
                    stats.other.push(m.model + ': HTTP ' + err.status);   // tham số không hợp với model này
                    continue outer;
                }
            } catch (e) {
                if (e.name !== 'AbortError') throw new Error('Không kết nối được Gemini: ' + e.message + ' (kiểm tra mạng)');
                stats.other.push(m.model + ': quá ' + (GEMINI_TEXT_TIMEOUT_MS / 1000) + ' giây');
                continue outer;
            } finally {
                clearTimeout(timer);
            }
        }
    }
    throw new Error(geminiFailMessage(stats, calls));
}

function spEsc(s) {
    // AI hay viết dấu gạch dài: đổi sang dấu phẩy cho dễ đọc
    return String(s == null ? '' : s).replace(/\s*—\s*/g, ', ').replace(/[&<>"']/g, function(c) {
        return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
}

function spNewLog() {
    return { v: 1, unlocked: 1, passed: [false, false, false], best: {}, mistakes: [], usage: null, mnemonic: '', attempts: 0, updated: 0 };
}
function spNormalizeLog(raw) {
    const log = spNewLog();
    if (!raw || typeof raw !== 'object') return log;
    log.unlocked = Math.min(3, Math.max(1, parseInt(raw.unlocked, 10) || 1));
    if (Array.isArray(raw.passed)) log.passed = [0, 1, 2].map(function(i) { return !!raw.passed[i]; });
    if (raw.best && typeof raw.best === 'object') log.best = raw.best;
    if (Array.isArray(raw.mistakes)) log.mistakes = raw.mistakes.filter(function(m) { return m && m.tag; });
    log.usage = raw.usage || null;
    log.mnemonic = raw.mnemonic || '';
    log.attempts = parseInt(raw.attempts, 10) || 0;
    log.updated = parseInt(raw.updated, 10) || 0;
    return log;
}
function spSortMistakes(list) {
    return list.sort(function(x, y) { return (y.n || 1) - (x.n || 1) || (y.at || 0) - (x.at || 0); });
}
// Gộp lịch sử từ field (đồng bộ từ máy tính) với lịch sử lưu trong máy này
function spMergeLogs(a, b) {
    a = spNormalizeLog(a); b = spNormalizeLog(b);
    const m = spNewLog();
    m.unlocked = Math.max(a.unlocked, b.unlocked);
    m.passed = [0, 1, 2].map(function(i) { return a.passed[i] || b.passed[i]; });
    Object.keys(Object.assign({}, a.best, b.best)).forEach(function(k) {
        const x = a.best[k], y = b.best[k];
        m.best[k] = !x ? y : !y ? x : ((y.at || 0) > (x.at || 0) ? y : x);
    });
    const byTag = {};
    a.mistakes.concat(b.mistakes).forEach(function(x) {
        const cur = byTag[x.tag];
        if (!cur) { byTag[x.tag] = Object.assign({}, x); return; }
        cur.n = Math.max(cur.n || 1, x.n || 1);
        if ((x.at || 0) > (cur.at || 0)) { cur.at = x.at; cur.label = x.label || cur.label; }
    });
    m.mistakes = spSortMistakes(Object.values(byTag)).slice(0, 8);
    m.usage = a.usage || b.usage;
    m.mnemonic = a.mnemonic || b.mnemonic;
    m.attempts = Math.max(a.attempts, b.attempts);
    m.updated = Math.max(a.updated, b.updated);
    return m;
}

    }
}
function spRecordMistakes(issues) {
    const st = spState, now = Date.now();
    const weak = spStoreGet('sp_weak_v1') || {};
    issues.forEach(function(i) {
        // Chỉ ghi lỗi ngôn ngữ (bỏ lỗi "sai đề"), mỗi đề chỉ đếm một lần cho mỗi loại lỗi
        if (i.severity !== 'error' || i.category === 'task' || st.taskTags[i.tag]) return;
        st.taskTags[i.tag] = true;
        const label = i.label_vi || i.fix || i.tag;
        const cur = st.log.mistakes.find(function(m) { return m.tag === i.tag; });
        if (cur) { cur.n = (cur.n || 1) + 1; cur.at = now; cur.label = label; }
        else st.log.mistakes.push({ tag: i.tag, label: label, cat: i.category, n: 1, at: now });
        const w = weak[i.tag] || { label: label, cat: i.category, n: 0 };
        w.n += 1; w.at = now; w.label = label;
        weak[i.tag] = w;
    });
    st.log.mistakes = spSortMistakes(st.log.mistakes).slice(0, 8);
    // Điểm yếu chung (mọi từ) chỉ lưu trong máy này, giữ 20 loại hay mắc nhất
    const top = Object.entries(weak).sort(function(a, b) { return b[1].n - a[1].n || (b[1].at || 0) - (a[1].at || 0); }).slice(0, 20);
    spStoreSet('sp_weak_v1', Object.fromEntries(top));
}

function spCleanUsage(u) {
    if (!u || typeof u !== 'object') return null;
    const pick = function(arr) {
        return (Array.isArray(arr) ? arr : []).filter(function(it) { return it && it.form; }).slice(0, 5).map(function(it) {
            return { form: String(it.form), note_vi: String(it.note_vi || ''), example: String(it.example || '') };
        });
    };
    const out = { particles: pick(u.particles), collocations: pick(u.collocations), structures: pick(u.structures), phrases: pick(u.phrases) };
    return (out.particles.length + out.collocations.length + out.structures.length + out.phrases.length) ? out : null;
}

}

', with pinyin in brackets' : '') + '), "note_vi" (short Vietnamese meaning or note) and an optional short "example".',
        '  "particles": ' + (zh ? 'particles / measure words / complements that go with it' : 'dependent prepositions / particles that go with it') + ' (max 4); "collocations": the most common collocations (max 5); "structures": key grammar patterns (max 3); "phrases": phrasal verbs, idioms or fixed phrases containing it (max 4). Use an empty array when nothing fits.',
        '- ALSO fill "mnemonic_vi": ONE short, vivid Vietnamese memory trick (max 40 words) linking the word\'s sound/shape/meaning with its key partner word.'
    ].join('\n');
}
function spBuildTaskPrompt(level, needUsage) {
    const st = spState, zh = st.lang === 'zh';
    const recent = spStoreGet('sp_recent_v1::' + st.word.toLowerCase()) || [];
    const parts = [
        'You are an expert ' + spLangName() + ' teacher for Vietnamese learners. Create ONE practice task that makes the learner actively USE the target word, so they remember it together with its prepositions/particles, collocations, grammar patterns and phrases.',
        '',
        spCardContext(),
        '',
        spLevelGuide(level) + spMistakeContext()
    ];
    if (recent.length) parts.push('\nDO NOT repeat or closely copy these recent tasks:\n' + recent.map(function(t) { return '- ' + t; }).join('\n'));
    parts.push(
        '',
        'RULES:',
        '- "task_vi": the task shown to the learner, in Vietnamese. It must NOT contain the answer, nor the required preposition/particle.',
        '- "phrase_vi": at level 1 the exact short Vietnamese phrase to translate (3–8 words, must appear word-for-word inside "task_vi" if task_vi is a sentence); at level 2–3 an empty string.',
        '- "focus": one of particle, collocation, grammar, phrase, what this task trains.',
        '- "focus_hint_vi": one short Vietnamese hint pointing to what to pay attention to, WITHOUT giving the answer.',
        '- "must_use": the words the answer must contain: always the target word; at level 3 also the one extra element. Never list the preposition/particle the learner has to find by themselves.',
        '- "reference_answers": 1–3 natural, correct answers.' + (zh ? ' Hanzi only, no pinyin; put the pinyin of the first answer in "reference_pinyin".' : ''),
        '- Use the meaning shown on the card. Make it natural and useful in real life.'
    );
    if (needUsage) parts.push(spUsageInstruction());
    parts.push('Return JSON only.');
    return parts.join('\n');
}
function spBuildGradePrompt(answer, attempt, needUsage) {
    const st = spState, zh = st.lang === 'zh', t = st.task;
    const expected = { 1: 'a short phrase (a full sentence containing the correct phrase is also fine)', 2: 'one sentence', 3: '1–2 sentences responding to the situation' };
    const parts = [
        'You are a strict but encouraging ' + spLangName() + ' teacher grading a Vietnamese learner. Explain in simple Vietnamese, without linguistic jargon.',
        '',
        spCardContext(),
        ''
    ];
    if (st.mode === 'free') {
        parts.push('MODE: free writing. the learner wrote their own sentence with the target word (no task). Judge whether the target word and its prepositions/particles, collocations, grammar and phrases are used correctly and naturally.');
    } else {
        parts.push(
            'MODE: ' + SP_LEVELS[String(st.mode)].name + '. Expected answer: ' + expected[st.mode] + '.',
            'TASK GIVEN (Vietnamese): ' + JSON.stringify(t.task_vi),
            t.phrase_vi ? 'PHRASE TO TRANSLATE (only this part is required): ' + JSON.stringify(t.phrase_vi) : '',
            'MUST USE: ' + JSON.stringify(t.must_use),
            'REFERENCE ANSWERS (for you only; other correct answers are fine): ' + JSON.stringify(t.reference_answers)
        );
    }
    parts.push(
        'LEARNER ANSWER: ' + JSON.stringify(answer),
        'ATTEMPT: ' + attempt + ' of ' + SP_MAX_ATTEMPTS,
        '',
        'CHECK in this order: (1) the target word has the right meaning and form; (2) prepositions/particles that go with it; (3) collocations and word choice; (4) grammar (tense/aspect, word order, articles, sentence structure); (5) phrasal verbs, idioms, fixed phrases if used or required; (6) the answer does what was asked' + (st.mode === 'free' ? '' : ' and keeps the meaning of the task') + '.',
        'RULES:',
        '- "issues": every real problem, max 4. "severity": "error" for anything wrong or unnatural to a native speaker; "suggestion" only for an optional improvement of an already correct answer.',
        '  "category": one of particle, collocation, grammar, phrase, word_choice, spelling, task. Use "task" only when the meaning does not match the task or a required word is missing, never for answer length or format.',
        '  "excerpt": the exact wrong part copied from the learner answer (empty if something is missing).',
        '  "hint_vi": a Vietnamese hint that helps the learner fix it by themselves WITHOUT giving the corrected words.',
        '  "fix": the corrected words. "explain_vi": 1–2 short Vietnamese sentences on why.',
        '  "tag": a short stable lowercase key for this kind of mistake, e.g. "prep:interested-in", "particle:le-completion", "colloc:make-decision", "grammar:present-perfect".',
        '  "label_vi": a short Vietnamese label for the learner\'s mistake list, e.g. "Sai giới từ: interested ON → IN".',
        '- "pass": true only when there is no issue with severity "error".',
        '- "summary_vi": one short Vietnamese sentence with the verdict. If the answer is not correct, do NOT reveal the corrected words.',
        '- "corrected": the learner answer with minimal corrections (keep their idea); the same as the answer if it is already correct.',
        '- "model_answer": one natural, idiomatic answer' + (zh ? ' (Hanzi only); put its pinyin in "model_pinyin" and the pinyin of "corrected" in "corrected_pinyin".' : '.'),
        '- "explain_vi": max 80 words in Vietnamese on the key usage points of the target word shown by this answer.',
        '- "upgrade_vi": if pass, one tip to sound more natural or advanced; otherwise an empty string.'
    );
    if (needUsage) parts.push(spUsageInstruction());
    parts.push('Return JSON only.');
    return parts.join('\n');
}
function spNormalizeGrade(d) {
    if (!d || typeof d !== 'object') throw new Error('AI trả về dữ liệu không hợp lệ');
    const issues = (Array.isArray(d.issues) ? d.issues : []).filter(function(i) { return i && (i.hint_vi || i.fix); }).slice(0, 5).map(function(i) {
        return {
            severity: i.severity === 'suggestion' ? 'suggestion' : 'error',
            category: String(i.category || ''),
            excerpt: String(i.excerpt || ''),
            hint_vi: String(i.hint_vi || ''),
            fix: String(i.fix || ''),
            explain_vi: String(i.explain_vi || ''),
            tag: String(i.tag || i.category || 'other').toLowerCase().slice(0, 60),
            label_vi: String(i.label_vi || '')
        };
    });
    const str = function(v) { return v == null ? '' : String(v); };
    return {
        // Không tin hẳn cờ pass: còn lỗi nặng thì chưa đạt
        pass: d.pass === true && !issues.some(function(i) { return i.severity === 'error'; }),
        issues: issues,
        summary_vi: str(d.summary_vi), corrected: str(d.corrected), corrected_pinyin: str(d.corrected_pinyin),
        model_answer: str(d.model_answer), model_pinyin: str(d.model_pinyin),
        explain_vi: str(d.explain_vi), upgrade_vi: str(d.upgrade_vi),
        raw: d
    };
}

// ---------- Giao diện ----------
function spDefaultLevel(log) {
    for (let l = 1; l <= log.unlocked; l++) if (!log.passed[l - 1]) return l;
    return log.unlocked;
}
function spInit() {
    const input = spEl('sp-input');
    if (!input) return;
    const word = spText('.word');
    if (!word) return;
    const fieldLog = spJson('#user-sentence-data');
    const storeKey = 'sp_log_v1::' + word.toLowerCase();
    const log = spMergeLogs(fieldLog && fieldLog.v ? fieldLog : null, spStoreGet(storeKey));
    spState = {
        token: Math.random().toString(36).slice(2), word: word, lang: /[㐀-鿿]/.test(word) ? 'zh' : 'en',
        storeKey: storeKey, log: log, mode: spDefaultLevel(log),
        task: null, attempts: 0, finished: false, lastGrade: null, taskTags: {}, busy: false
    };
    input.addEventListener('keydown', function(e) {
        if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) { e.preventDefault(); spCheck(); }
    });
    input.addEventListener('focus', spOnInputFocus);
    spEl('sp-container').addEventListener('click', function(e) {
        const btn = e.target.closest('[data-sp-say]');
        if (btn) speakTextWithEdge(btn.getAttribute('data-sp-say'), e);
    });
    spRenderLevels();
    spRenderMemory();
    spRenderTask();
    spUpdateButtons();
}
function spRenderLevels() {
    const st = spState;
    let html = '';
    [1, 2, 3].forEach(function(l) {
        const locked = l > st.log.unlocked, passed = st.log.passed[l - 1];
        html += '<button type="button" class="sp-level' + (st.mode === l ? ' active' : '') + (locked ? ' locked' : '') + '"'
            + (locked ? ' disabled title="Đạt bậc trước để mở"' : '') + ' onclick="spSetMode(' + l + ')">'
            + (passed ? '✅' : locked ? '🔒' : '⭐') + ' ' + SP_LEVELS[l].name + '<small>' + SP_LEVELS[l].short + '</small></button>';
    });
    html += '<button type="button" class="sp-level' + (st.mode === 'free' ? ' active' : '') + '" onclick="spSetMode(\'free\')">✍️ '
        + SP_LEVELS.free.name + '<small>' + SP_LEVELS.free.short + '</small></button>';
    spEl('sp-levels').innerHTML = html;
}
function spRenderMemory() {
    const st = spState, box = spEl('sp-memory');
    const best = st.log.best['3'] || st.log.best['2'] || st.log.best['1'] || st.log.best.free;
    const mistakes = st.log.mistakes.slice(0, 3);
    if (!best && !mistakes.length) { box.style.display = 'none'; box.innerHTML = ''; return; }
    let html = '';
    if (best) {
        html += '<div class="sp-mem-label">🏆 Câu tốt nhất của bạn (' + SP_LEVELS[best.level].name + ')</div>'
            + '<div class="sp-mem-text">' + spEsc(best.text) + ' ' + spSayBtn(best.text) + '</div>';
    }
    if (mistakes.length) {
        html += '<div class="sp-mem-mistakes"><div class="sp-mem-label">⚠️ Lỗi bạn hay mắc với từ này</div><ul>'
            + mistakes.map(function(m) { return '<li>' + spEsc(m.label) + (m.n > 1 ? ' <span class="sp-count">×' + m.n + '</span>' : '') + '</li>'; }).join('')
            + '</ul></div>';
    }
    box.innerHTML = html;
    box.style.display = 'block';
}
function spRenderTask() {
    const st = spState, box = spEl('sp-task'), input = spEl('sp-input');
    // Không khoá ô nhập (trên điện thoại ô bị khoá thì chạm vào không hiện bàn phím);
    // chạm vào ô lúc chưa có đề / đã xong sẽ tự ra đề tiếp (xem spOnInputFocus)
    input.placeholder = st.finished
        ? (st.mode === 'free' ? 'Chạm để viết câu khác...' : 'Chạm để làm đề tiếp theo...')
        : (st.mode !== 'free' && !st.task ? 'Chạm vào đây để AI ra đề, rồi viết câu trả lời...' : 'Viết câu trả lời của bạn ở đây...');
    if (st.mode === 'free') {
        box.innerHTML = '<div class="sp-task-label">✍️ Tự do</div><div class="sp-task-text">Tự đặt một câu bất kỳ có dùng "<b>'
            + spEsc(st.word) + '</b>". Câu và lỗi vẫn được lưu nhưng không tính lên bậc.</div>';
        return;
    }
    const label = SP_LEVELS[String(st.mode)].name + ' · ' + SP_LEVELS[String(st.mode)].short;
    if (!st.task) {
        box.innerHTML = '<div class="sp-task-label">' + label + '</div><div class="sp-task-text sp-muted">Bấm <b>🎲 Đề mới</b> hoặc chạm vào ô bên dưới để AI ra đề.</div>';
        return;
    }
    const t = st.task;
    const progress = st.finished ? 'Đã xong' : 'Lần thử ' + (st.attempts + 1) + '/' + SP_MAX_ATTEMPTS;
    box.innerHTML = '<div class="sp-task-label">' + label + ' · ' + progress + '</div>'
        + spTaskTextHtml(t)
        + (t.focus_hint_vi ? '<div class="sp-task-hint">💡 ' + spFmt(t.focus_hint_vi) + '</div>' : '')
        + (t.must_use.length ? '<div class="sp-task-must">Phải dùng: ' + t.must_use.map(function(w) { return '<code>' + spEsc(w) + '</code>'; }).join(' ') + '</div>' : '');
}
// Bậc 1: hiện rõ CỤM cần dịch; câu dài (nếu có) chỉ là ngữ cảnh, tô màu đúng cụm đó trong câu
function spTaskTextHtml(t) {
    const phrase = t.phrase_vi;
    const lang = spState.lang === 'zh' ? 'tiếng Trung' : 'tiếng Anh';
    if (!phrase) {
        // Bậc 2: dịch cả câu; Bậc 3: trả lời tình huống
        const lead = spState.mode === 2 ? 'Viết câu ' + lang + ' cho:'
            : spState.mode === 3 ? 'Trả lời bằng 1–2 câu ' + lang + ':' : '';
        return (lead ? '<div class="sp-task-lead">' + lead + '</div>' : '') + '<div class="sp-task-text">' + spFmt(t.task_vi) + '</div>';
    }
    let html = '<div class="sp-task-text">Viết cụm ' + lang + ' cho:</div><div class="sp-phrase">' + spEsc(phrase) + '</div>';
    const context = (t.task_vi || '').trim();
    // Chỉ hiện ngữ cảnh khi là câu thật (không phải câu lệnh "Hãy dịch..." hay chỉ lặp lại cụm)
    const rest = context.toLowerCase().replace(phrase.toLowerCase(), '').replace(/[\s.,:;!?"'“”]+/g, ' ').trim();
    const isInstruction = /(hãy|dịch|cụm từ sau|sang tiếng)/i.test(rest);
    if (context && rest.split(' ').length >= 3 && !isInstruction) {
        const pos = context.toLowerCase().indexOf(phrase.toLowerCase());
        const marked = pos >= 0
            ? spEsc(context.slice(0, pos)) + '<mark class="sp-phrase-mark">' + spEsc(context.slice(pos, pos + phrase.length)) + '</mark>' + spEsc(context.slice(pos + phrase.length))
            : spEsc(context);
        html += '<div class="sp-context">Ngữ cảnh: ' + marked + '</div>';
    }
    return html;
}
function spOnInputFocus() {
    const st = spState, input = spEl('sp-input');
    if (!st) return;
    // Bàn phím điện thoại hay che mất ô nhập -> cuộn ô lên giữa màn hình
    // (iPhone tự cuộn ô lên; chỉ cuộn thêm khi ô thật sự nằm dưới mép bàn phím, tránh giật màn hình)
    setTimeout(function() {
        try {
            const visibleBottom = window.visualViewport ? window.visualViewport.height : window.innerHeight;
            if (input.getBoundingClientRect().bottom > visibleBottom) input.scrollIntoView({ block: 'center' });
        } catch (e) { /* trình duyệt cũ */ }
    }, 400);
    if (st.busy) return;
    if (st.finished) {
        if (st.mode === 'free') spResetFree();
        else spNewTask();
    } else if (st.mode !== 'free' && !st.task) {
        spNewTask();
    }
}
function spUpdateButtons() {
    const st = spState;
    const btnNew = spEl('sp-btn-new'), btnCheck = spEl('sp-btn-check'), btnReveal = spEl('sp-btn-reveal');
    btnNew.style.display = st.mode === 'free' ? 'none' : '';
    btnNew.disabled = st.busy;
    btnNew.textContent = st.task ? '🎲 Đề khác' : '🎲 Đề mới';
    btnCheck.disabled = st.busy || st.finished || (st.mode !== 'free' && !st.task);
    btnCheck.textContent = st.attempts > 0 && !st.finished ? '🔁 Chấm lại' : '✅ Chấm';
    const canReveal = !st.finished && ((st.task && st.task.reference_answers.length) || st.lastGrade);
    btnReveal.disabled = st.busy || !canReveal;
}
function spSetBusy(busy, text) {
    const st = spState, box = spEl('sp-status');
    st.busy = busy;
    if (busy) {
        box.innerHTML = '<span class="sp-spinner"></span><span>' + spEsc(text || 'Đang xử lý...') + '</span>';
        box.style.display = 'flex';
    } else {
        box.style.display = 'none';
        box.innerHTML = '';
    }
    spUpdateButtons();
}
function spIssueHtml(i, revealed) {
    const cat = SP_CATEGORY_NAMES[i.category] || i.category || 'Lỗi';
    let excerpt = '';
    if (i.excerpt) excerpt = '“' + spEsc(i.excerpt) + '”' + (revealed && i.fix ? ' → <b>' + spEsc(i.fix) + '</b>' : '');
    else if (revealed && i.fix) excerpt = '→ <b>' + spEsc(i.fix) + '</b>';
    return '<div class="sp-issue ' + (i.severity === 'error' ? 'error' : 'suggestion') + '">'
        + '<div class="sp-issue-cat">' + (i.severity === 'error' ? '' : 'Gợi ý · ') + spEsc(cat) + '</div>'
        + (excerpt ? '<div class="sp-issue-excerpt">' + excerpt + '</div>' : '')
        + '<div class="sp-issue-hint">' + spFmt(revealed ? (i.explain_vi || i.hint_vi) : i.hint_vi) + '</div>'
        + '</div>';
}
function spAnswersHtml(g, t) {
    const row = function(label, text, pinyin) {
        if (!text) return '';
        return '<div class="sp-answer"><div class="sp-answer-label">' + label + '</div><div class="sp-answer-text">' + spEsc(text) + ' ' + spSayBtn(text) + '</div>'
            + (pinyin ? '<div class="item-content-pinyin">' + spEsc(pinyin) + '</div>' : '') + '</div>';
    };
    let html = '';
    if (g && !g.pass && g.corrected) html += row('✏️ Câu của bạn sau khi sửa', g.corrected, g.corrected_pinyin);
    if (g && g.model_answer) {
        html += row(g.pass ? '✨ Câu mẫu tham khảo' : '✅ Câu mẫu', g.model_answer, g.model_pinyin);
    } else if (t && t.reference_answers.length) {
        html += t.reference_answers.map(function(a, k) { return row(k ? '🔁 Cách nói khác' : '✅ Đáp án', a, k ? '' : t.reference_pinyin); }).join('');
    }
    return html;
}
function spUsageHtml(u) {
    const sec = function(title, items) {
        if (!items || !items.length) return '';
        return '<div class="sp-usage-sec"><div class="sp-usage-title">' + title + '</div>' + items.map(function(it) {
            return '<div class="sp-usage-item"><b>' + spEsc(it.form) + '</b>' + (it.note_vi ? ': ' + spEsc(it.note_vi) : '')
                + (it.example ? '<div class="sp-usage-ex">' + spEsc(it.example) + '</div>' : '') + '</div>';
        }).join('') + '</div>';
    };
    return '<div class="sp-usage"><div class="sp-usage-head">📇 Thẻ tóm tắt cách dùng "' + spEsc(spState.word) + '"</div>'
        + sec(spState.lang === 'zh' ? '🔗 Trợ từ / lượng từ / bổ ngữ đi kèm' : '🔗 Giới từ / tiểu từ đi kèm', u.particles)
        + sec('🤝 Collocation', u.collocations)
        + sec('🏗️ Cấu trúc', u.structures)
        + sec('🚀 Phrasal verb / idiom / cụm cố định', u.phrases)
        + '</div>';
}
function spMoreHtml(g) {
    const st = spState;
    let html = '';
    if (g && g.explain_vi) html += '<div class="sp-explain">📝 ' + spFmt(g.explain_vi) + '</div>';
    if (g && g.pass && g.upgrade_vi) html += '<div class="sp-upgrade">🚀 <b>Nâng cao:</b> ' + spFmt(g.upgrade_vi) + '</div>';
    if (st.log.usage) html += spUsageHtml(st.log.usage);
    if (st.log.mnemonic) html += '<div class="sp-mnemonic">🧠 <b>Mẹo nhớ:</b> ' + spFmt(st.log.mnemonic) + '</div>';
    return html;
}
function spRenderResult(opts) {
    opts = opts || {};
    const st = spState, g = st.lastGrade, t = st.task;
    const revealed = st.finished;
    let html = '';
    if (g) {
        let head;
        if (g.pass) head = '🎉 Đạt!';
        else if (opts.gaveUp) head = '📘 Đáp án';
        else if (revealed) head = '📘 Chưa đạt sau ' + st.attempts + ' lần, xem đáp án bên dưới';
        else head = '🔁 Chưa đạt, tự sửa lại nhé (còn ' + (SP_MAX_ATTEMPTS - st.attempts) + ' lần)';
        html += '<div class="sp-verdict ' + (g.pass ? 'pass' : 'fail') + '"><div class="sp-verdict-head">' + head + '</div>'
            + (g.summary_vi ? '<div>' + spFmt(g.summary_vi) + '</div>' : '') + '</div>';
        if (opts.unlockedNow) html += '<div class="sp-unlock">🔓 Đã mở ' + SP_LEVELS[opts.unlockedNow].name + ': ' + SP_LEVELS[opts.unlockedNow].short + '!</div>';
        if (opts.allDone) html += '<div class="sp-unlock">🏆 Bạn đã vượt cả 3 bậc với từ này!</div>';
        html += g.issues.filter(function(i) { return i.severity === 'error'; }).map(function(i) { return spIssueHtml(i, revealed); }).join('');
        if (revealed) html += g.issues.filter(function(i) { return i.severity !== 'error'; }).map(function(i) { return spIssueHtml(i, true); }).join('');
    } else if (opts.gaveUp) {
        html += '<div class="sp-verdict fail"><div class="sp-verdict-head">📘 Đáp án</div></div>';
    }
    if (revealed) {
        html += spAnswersHtml(g, t);
        const more = spMoreHtml(g);
        if (more) {
            html += '<button type="button" class="toggle-section-btn" onclick="toggleSection(\'sp-more\')">📖 Xem thêm: giải thích, thẻ cách dùng, mẹo nhớ ▼</button>'
                + '<div id="sp-more" class="collapsible-section" style="display: none;">' + more + '</div>';
        }
        html += st.mode === 'free'
            ? '<button type="button" class="sp-btn sp-btn-next" onclick="spResetFree()">✍️ Viết câu khác</button>'
            : '<button type="button" class="sp-btn sp-btn-next" onclick="spNewTask()">🎲 Đề tiếp theo</button>';
    }
    spEl('sp-result').innerHTML = html;
}
function spResetTask() {
    const st = spState;
    st.attempts = 0; st.finished = false; st.lastGrade = null; st.taskTags = {};
    spEl('sp-input').value = '';
    spEl('sp-result').innerHTML = '';
}

// ---------- Hành động ----------
function spSetMode(mode) {
    const st = spState;
    if (!st || st.busy || st.mode === mode) return;
    if (mode !== 'free' && mode > st.log.unlocked) return;
    st.mode = mode;
    st.task = null;
    spResetTask();
    spRenderLevels();
    spRenderTask();
    spUpdateButtons();
    if (mode !== 'free') spNewTask();
}
function spResetFree() {
    const st = spState;
    if (!st || st.busy) return;
    spResetTask();
    spRenderTask();
    spUpdateButtons();
    spEl('sp-input').focus();
}
async function spNewTask() {
    const st = spState;
    if (!st || st.busy || st.mode === 'free') return;
    const token = st.token, level = st.mode, needUsage = !st.log.usage;
    const textAtStart = spEl('sp-input').value;
    spSetBusy(true, 'Đang ra đề ' + SP_LEVELS[String(level)].name + '...');
    try {
        const data = await geminiGenerateJson(spBuildTaskPrompt(level, needUsage), spSchema(SP_TASK_SCHEMA, needUsage), 0.9);
        if (!spIsCurrent(token)) return;
        if (!data || !data.task_vi) throw new Error('AI không trả về đề bài');
        const list = function(v) { return Array.isArray(v) ? v.map(String).filter(Boolean) : []; };
        st.task = {
            task_vi: String(data.task_vi),
            // Bậc 1: chỉ dịch đúng cụm này (AI hay đưa cả câu dài làm người học không biết dịch phần nào)
            phrase_vi: level === 1 ? String(data.phrase_vi || '').trim() : '',
            focus: String(data.focus || ''),
            focus_hint_vi: String(data.focus_hint_vi || ''),
            // Bậc 1–2 chỉ hiện từ đang học: AI hay liệt kê cả từ đi kèm, làm lộ đáp án
            must_use: level === 3 && list(data.must_use).length ? list(data.must_use).slice(0, 3) : [st.word],
            reference_answers: list(data.reference_answers),
            reference_pinyin: String(data.reference_pinyin || '')
        };
        if (needUsage) { spStoreUsage(data); if (st.log.usage) spSaveLog(); }
        spRememberTask(st.task.task_vi);
        // Người học đã gõ trong lúc chờ đề -> giữ lại chữ đó, không xoá
        const typed = spEl('sp-input').value;
        spResetTask();
        if (typed !== textAtStart) spEl('sp-input').value = typed;
        spRenderTask();
        spEl('sp-input').focus();
    } catch (e) {
        if (spIsCurrent(token)) { console.error('[Luyện câu]', e); showPracticeMessage('❌ ' + e.message, 'error'); }
    } finally {
        if (spIsCurrent(token)) spSetBusy(false);
    }
}
async function spCheck() {
    const st = spState;
    if (!st || st.busy || st.finished) return;
    if (st.mode !== 'free' && !st.task) return;
    const answer = spEl('sp-input').value.trim();
    if (!answer) { showPracticeMessage('⚠️ Hãy viết câu trả lời trước khi chấm', 'error'); return; }
    const token = st.token, attempt = st.attempts + 1, needUsage = !st.log.usage;
    spSetBusy(true, 'Đang chấm...');
    try {
        const g = spNormalizeGrade(await geminiGenerateJson(spBuildGradePrompt(answer, attempt, needUsage), spSchema(SP_GRADE_SCHEMA, needUsage), 0.2));
        if (!spIsCurrent(token)) return;
        st.attempts = attempt;
        st.lastGrade = g;
        st.log.attempts += 1;
        if (needUsage) spStoreUsage(g.raw);
        spRecordMistakes(g.issues);
        const opts = {};
        if (g.pass) {
            st.finished = true;
            st.log.best[String(st.mode)] = { level: String(st.mode), text: answer, at: Date.now() };
            if (st.mode !== 'free') {
                const wasPassed = st.log.passed[st.mode - 1];
                st.log.passed[st.mode - 1] = true;
                if (st.mode < 3 && st.log.unlocked < st.mode + 1) { st.log.unlocked = st.mode + 1; opts.unlockedNow = st.mode + 1; }
                opts.allDone = !wasPassed && st.log.passed.every(Boolean);
            }
        } else if (attempt >= SP_MAX_ATTEMPTS) {
            st.finished = true;
        }
        spSaveLog();
        spRenderResult(opts);
        spRenderLevels();
        spRenderMemory();
        spRenderTask();
    } catch (e) {
        if (spIsCurrent(token)) { console.error('[Luyện câu]', e); showPracticeMessage('❌ ' + e.message, 'error'); }
    } finally {
        if (spIsCurrent(token)) spSetBusy(false);
    }
}
function spReveal() {
    const st = spState;
    if (!st || st.busy || st.finished) return;
    st.finished = true;
    spRenderResult({ gaveUp: true });
    spRenderTask();
    spUpdateButtons();
}

var RD_MAX_WORDS = 5;
var RD_MAX_STRUCTURES = 2;
var RD_VOCAB_SCHEMA = {
    type: 'OBJECT',
    properties: {
        vocabulary: {
            type: 'ARRAY',
            items: {
                type: 'OBJECT',

                },

                },
                required: ['pattern', 'explain_vi'],
                propertyOrdering: ['pattern', 'in_text', 'explain_vi', 'example', 'example_vi']
            }

// Chuẩn hoá dữ liệu từ vựng (bản do thẻ tạo hoặc do các nút trong add-on tạo, tên khoá khác nhau)
function rdNormalizeData(d) {
    const s = function(v) { return v == null ? '' : String(v); };
    const vocabulary = (Array.isArray(d.vocabulary) ? d.vocabulary : []).filter(function(v) { return v && (v.term || v.word); }).slice(0, RD_MAX_WORDS).map(function(v) {
        return {
            term: s(v.term || v.base_form || v.word), in_text: s(v.in_text || v.word), pos: s(v.pos),
            meaning_vi: s(v.meaning_vi), example: s(v.example || v.example_from_reading), example_vi: s(v.example_vi || v.translation)
        };
    });
    const structures = (Array.isArray(d.structures) ? d.structures : []).filter(function(x) { return x && (x.pattern || x.structure); }).slice(0, RD_MAX_STRUCTURES).map(function(x) {
        return {
            pattern: s(x.pattern || x.structure), in_text: s(x.in_text), explain_vi: s(x.explain_vi || x.explanation_vi),
            example: s(x.example || x.example_from_reading), example_vi: s(x.example_vi || x.translation)
        };
    });
    return { v: 1, at: d.at || Date.now(), vocabulary: vocabulary, structures: structures };
}
// Lấy bản mới hơn giữa field (đồng bộ từ máy tính) và bản lưu trong máy này.
// Field chứa HTML kiểu cũ thì giữ lại để xem trong mục gập.
function rdLoadData() {
    const st = rdState, fieldEl = spEl('rd-vocab-field');
    const raw = fieldEl ? (fieldEl.textContent || '').trim() : '';
    let field = null;
    if (raw) {
        try { const d = JSON.parse(raw); if (d && d.v === 1) field = d; } catch (e) { /* HTML kiểu cũ */ }
    }
    const best = [field, spStoreGet(st.storeKey)].filter(Boolean).sort(function(a, b) { return (b.at || 0) - (a.at || 0); })[0];
    if (best) return rdNormalizeData(best);
    if (raw) {
        const structEl = spEl('rd-struct-field');
        st.legacyHtml = fieldEl.innerHTML + (structEl ? structEl.innerHTML : '');
    }
    return null;
}
// Tô từ khoá và gạch chân các cụm từ vựng (bấm để xem nghĩa) trong bài đọc
function rdMarkText(text) {
    const st = rdState, lower = text.toLowerCase(), ranges = [];
    if (st.data) {
        st.data.vocabulary.forEach(function(v, i) {
            const needle = (v.in_text || v.term).trim().toLowerCase();
            const pos = needle.length > 1 ? lower.indexOf(needle) : -1;
            if (pos >= 0) ranges.push({ start: pos, end: pos + needle.length, kind: 'term', idx: i });
        });
    }
    const kw = st.word.trim();
    if (kw) {
        const esc = kw.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
        // Tiếng Anh: bắt cả dạng biến đổi (plan -> plans, planned)
        const re = /^[a-z]/i.test(kw) ? new RegExp('\\b' + esc + '\\w*', 'gi') : new RegExp(esc, 'g');
        let m;
        while ((m = re.exec(text))) ranges.push({ start: m.index, end: m.index + m[0].length, kind: 'key' });
    }
    ranges.sort(function(a, b) { return a.start - b.start || (b.end - b.start) - (a.end - a.start); });
    const kept = [];
    ranges.forEach(function(r) {
        const last = kept[kept.length - 1];
        if (last && r.start < last.end) {
            // Chồng lấn: giữ cụm từ vựng (bấm được); nếu cụm chứa từ khoá thì tô cả hai
            if (last.kind === 'term' && r.kind === 'key') last.hasKey = true;
            else if (last.kind === 'key' && r.kind === 'term') { r.hasKey = true; kept[kept.length - 1] = r; }
            return;
        }
        kept.push(r);
    });
    let html = '', cur = 0;
    kept.forEach(function(r) {
        if (r.start < cur) return;
        html += spEsc(text.slice(cur, r.start));
        const seg = spEsc(text.slice(r.start, r.end));
        html += r.kind === 'term'
            ? '<span class="rd-term tappable' + (r.hasKey ? ' rd-key' : '') + '" data-rd-i="' + r.idx + '">' + seg + '</span>'
            : '<mark class="rd-key">' + seg + '</mark>';
        cur = r.end;
    });
    return (html + spEsc(text.slice(cur))).replace(/\n/g, '<br>');
}

function rdInit() {
    if (!spEl('rd-reading')) return;
    const word = spText('.word');
    rdState = {
        token: Math.random().toString(36).slice(2), word: word, reading: rdParseReading(),
        storeKey: 'rv_v1::' + word.toLowerCase(), data: null, legacyHtml: '', busy: false
    };
    rdState.data = rdLoadData();
    spEl('rd-container').addEventListener('click', function(e) {
        const say = e.target.closest('[data-sp-say]');
        if (say) { speakTextWithEdge(say.getAttribute('data-sp-say'), e); return; }
        const term = e.target.closest('.rd-term');
        if (term) rdShowPop(term);
        else rdHidePop();
    });
    rdRenderReading();
    rdRenderQuestions();
    rdRenderVocab();
}
function rdRenderReading() {
    const r = rdState.reading, box = spEl('rd-reading');
    if (!r) {
        box.innerHTML = '<div class="rd-empty">Thẻ này chưa có bài đọc.</div>';
        return;
    }
    box.innerHTML = '<div class="rd-passage" id="rd-passage">'
        + (r.title ? '<div class="rd-title">' + spEsc(rdPlainText(r.title)) + '</div>' : '')
        + '<div class="rd-text">' + rdMarkText(rdPlainText(r.content)) + '</div>'
        + '<div class="rd-pop tappable" id="rd-pop" style="display: none;"></div>'
        + '</div>'
        + (r.translation
            ? '<button type="button" class="toggle-section-btn rd-toggle" onclick="toggleSection(\'rd-translation\')">🌐 Xem bản dịch ▼</button>'
              + '<div id="rd-translation" class="rd-translation" style="display: none;">' + spFmt(rdPlainText(r.translation)) + '</div>'
            : '');
}
function rdShowPop(term) {
    const v = rdState.data && rdState.data.vocabulary[Number(term.getAttribute('data-rd-i'))];
    const pop = spEl('rd-pop'), passage = spEl('rd-passage');
    if (!v || !pop || !passage) return;
    pop.innerHTML = '<b>' + spEsc(v.term) + '</b>' + (v.pos ? ' <span class="rd-pos">' + spEsc(v.pos) + '</span>' : '')
        + '<div>' + spEsc(v.meaning_vi) + '</div>';
    pop.style.display = 'block';
    const left = Math.max(0, Math.min(term.offsetLeft, passage.clientWidth - pop.offsetWidth));
    pop.style.left = left + 'px';
    pop.style.top = (term.offsetTop + term.offsetHeight + 4) + 'px';
}
function rdHidePop() {
    const pop = spEl('rd-pop');
    if (pop) pop.style.display = 'none';
}
function rdRenderQuestions() {
    const r = rdState.reading, box = spEl('rd-questions');
    const qs = (r && Array.isArray(r.questions)) ? r.questions.filter(function(q) { return q && q.question; }) : [];
    if (!qs.length) { box.innerHTML = ''; return; }
    box.innerHTML = '<div class="rd-section-title">❓ Câu hỏi đọc hiểu</div>' + qs.map(function(q, qi) {
        const options = Array.isArray(q.options) ? q.options : [];
        return '<div class="rd-q" id="rd-q-' + qi + '"><div class="rd-q-text">' + (qi + 1) + '. ' + spEsc(rdPlainText(q.question)) + '</div>'
            + options.map(function(o, oi) {
                return '<button type="button" class="rd-opt" onclick="rdAnswer(' + qi + ',' + oi + ')">' + String.fromCharCode(65 + oi) + '. ' + spEsc(rdPlainText(o)) + '</button>';
            }).join('')
            + '<div class="rd-q-fb" id="rd-q-fb-' + qi + '"></div></div>';
    }).join('');
    rdState.questions = qs;
}
function rdAnswer(qi, oi) {
    const q = rdState.questions && rdState.questions[qi], box = spEl('rd-q-' + qi);
    if (!q || !box || box.getAttribute('data-done')) return;
    box.setAttribute('data-done', '1');
    const correct = Number(q.correctAnswer);
    box.querySelectorAll('.rd-opt').forEach(function(b, k) {
        b.disabled = true;
        if (k === correct) b.classList.add('right');
        else if (k === oi) b.classList.add('wrong');
    });
    spEl('rd-q-fb-' + qi).innerHTML = (oi === correct ? '✅ Chính xác! ' : '❌ Chưa đúng. ') + (q.explanation ? spFmt(rdPlainText(q.explanation)) : '');
}
function rdExampleHtml(example, exampleVi) {

    return '<div class="rd-ex">' + (example ? '<div>' + spEsc(example) + ' ' + spSayBtn(example) + '</div>' : '')
        + (exampleVi ? '<div class="rd-ex-vi">' + spEsc(exampleVi) + '</div>' : '') + '</div>';
}
function rdRenderVocab() {
    const st = rdState, list = spEl('rd-vocab-list'), btn = spEl('rd-vocab-btn');
    if (!st.reading) { spEl('rd-vocab').style.display = 'none'; return; }
    btn.disabled = st.busy;
    if (st.busy) btn.innerHTML = '<span class="sp-spinner"></span> Đang tạo...';
    else btn.textContent = st.data ? '🔄 Tạo lại' : (st.legacyHtml ? '✨ Tạo bản gọn' : '✨ Tạo từ vựng');
    if (st.data) {
        const d = st.data;
        list.innerHTML = d.vocabulary.map(function(v) {
            return '<details class="rd-item"><summary class="tappable"><b>' + spEsc(v.term) + '</b>' + (v.pos ? ' <span class="rd-pos">' + spEsc(v.pos) + '</span>' : '')
                + (v.meaning_vi ? ': ' + spEsc(v.meaning_vi) : '') + '</summary>' + rdExampleHtml(v.example, v.example_vi) + '</details>';
        }).join('')
            + (d.structures.length ? '<div class="rd-sub-title">🏗️ Cấu trúc</div>' + d.structures.map(function(s) {
                return '<details class="rd-item"><summary class="tappable"><b>' + spEsc(s.pattern) + '</b>' + (s.explain_vi ? ': ' + spEsc(s.explain_vi) : '')
                    + '</summary>' + rdExampleHtml(s.example, s.example_vi) + '</details>';
            }).join('') : '');
    } else if (st.legacyHtml) {
        list.innerHTML = '<details class="rd-item"><summary class="tappable">Xem từ vựng đã tạo trước đây (bản cũ)</summary><div class="rd-legacy">' + st.legacyHtml + '</div></details>';
    } else {
        list.innerHTML = '<div class="rd-hint">AI chọn 5 cụm từ + 2 cấu trúc đáng học trong bài. Sau đó bấm vào cụm được gạch chân trong bài để xem nghĩa.</div>';
    }
}
function rdBuildPrompt() {
    const st = rdState, r = st.reading, content = rdPlainText(r.content);
    const zh = /[㐀-鿿]/.test(content);
    return [
        'You are a ' + (zh ? 'Chinese' : 'English') + ' teacher for Vietnamese learners. From the reading passage below, choose what is most worth learning so the learner remembers the target word "' + st.word + '" and can reuse the language of the passage.',
        'VIETNAMESE MEANING OF THE TARGET WORD: ' + (spText('.definition-vi') || '(not given)'),
        'PASSAGE TITLE: ' + rdPlainText(r.title),
        'PASSAGE:\n"""\n' + content + '\n"""',
        '',
        'RULES:',
        '- "vocabulary": exactly ' + RD_MAX_WORDS + ' items that appear in the passage. Prefer multi-word phrases and collocations (e.g. "draw up a plan", "long-term goals"), especially ones containing or naturally going with the target word; then useful single words. Do not pick the bare target word alone or very easy words.',
        '  Each item: "term" (dictionary form, e.g. "draw up"' + (zh ? ', followed by pinyin in brackets' : '') + '), "in_text" (the EXACT words copied from the passage, same spelling), "pos" (short, e.g. n, v, adj, adv, phr v, colloc, idiom), "meaning_vi" (short Vietnamese meaning in this context), "example" (the sentence from the passage that contains it), "example_vi" (Vietnamese translation of that sentence).',
        '- "structures": exactly ' + RD_MAX_STRUCTURES + ' useful grammar patterns used in the passage. Each: "pattern" (e.g. "be responsible for + V-ing"), "in_text" (exact words from the passage), "explain_vi" (one short Vietnamese sentence on how to use it), "example" (the sentence from the passage), "example_vi" (its Vietnamese translation).',
        'Return JSON only.'
    ].join('\n');
}
async function rdCreateVocab() {
    const st = rdState;
    if (!st || st.busy || !st.reading) return;
    const token = st.token;
    st.busy = true;
    rdRenderVocab();
    try {
        const d = rdNormalizeData(await geminiGenerateJson(rdBuildPrompt(), RD_VOCAB_SCHEMA, 0.3));
        if (!rdIsCurrent(token)) return;
        if (!d.vocabulary.length) throw new Error('AI không chọn được từ vựng nào, thử lại nhé');
        d.at = Date.now();
        st.data = d;
        st.legacyHtml = '';
        spStoreSet(st.storeKey, d);
        if (hasAnkiBridge()) {
            pycmd('rv_save:' + JSON.stringify({ word: st.word, data: d }), function(res) {
                if (res !== 'ok') console.warn('[Áp dụng] Không ghi được vào field Reading_Vocabulary:', res);
            });
        }
        rdRenderReading();   // gạch chân các cụm vừa tạo trong bài
    } catch (e) {
        if (rdIsCurrent(token)) { console.error('[Áp dụng]', e); showPracticeMessage('❌ ' + e.message, 'error'); }
    } finally {
        if (rdIsCurrent(token)) { st.busy = false; rdRenderVocab(); }
    }
}

/* ==================== EXERCISE FUNCTIONS - ĐÃ SỬA ==================== */
function checkAnswer(index, correctAnswer) {
    const inputElement = document.getElementById(`input-${index}`);
    const exerciseItem = document.getElementById(`exercise-item-${index}`);
    const explanationElement = document.getElementById(`explanation-${index}`);
    
    if (!inputElement || !exerciseItem) return;
    
    const userAnswer = inputElement.value.trim();
    
    // Xóa lớp cũ
    exerciseItem.classList.remove('correct', 'incorrect');
    
    if (userAnswer.toLowerCase() === correctAnswer.toLowerCase()) {
        exerciseItem.classList.add('correct');
    } else {
        exerciseItem.classList.add('incorrect');
    }
    
    // Hiển thị giải thích
    if (explanationElement) {
        explanationElement.style.display = 'block';
    }
}

function showAnswer(index, correctAnswer) {
    const inputElement = document.getElementById(`input-${index}`);
    const exerciseItem = document.getElementById(`exercise-item-${index}`);
    const explanationElement = document.getElementById(`explanation-${index}`);
    
    if (inputElement) {
        inputElement.value = correctAnswer;
    }
    
    if (exerciseItem) {
        exerciseItem.classList.remove('incorrect');
        exerciseItem.classList.add('correct');
    }
    
    if (explanationElement) {
        explanationElement.style.display = 'block';
    }
}

function displayChineseStory() {
    const el = document.getElementById('radicals-data');
    const storyBox = document.getElementById('chinese-story-container');
    const storyContent = document.getElementById('chinese-story-content');
    const raw = (el?.textContent || '').trim();
    if (!raw || !/^[\[{]/.test(raw)) { storyBox.style.display = 'none'; return; }

    try {
        const arr = JSON.parse(raw);
        if (!Array.isArray(arr)) { storyBox.style.display = 'none'; return; }
        let html = '';
        arr.forEach(item => {
            if (item?.type === 'memory_story' && item.content?.trim()) {
                html += `<div class="character-story">
                    <div class="story-text">
                        <div style="font-weight:bold;color:#e74c3c;margin-bottom:10px;">📖 Câu chuyện chữ Hán:</div>
                        ${item.content}
                    </div>
                </div><div class="story-separator">• • •</div>`;
            }
        });
        if (html) {
            html = html.replace(/<div class="story-separator">• • •<\/div>$/, '');
            storyContent.innerHTML = html;
            storyBox.style.display = 'block';
        } else storyBox.style.display = 'none';
    } catch { storyBox.style.display = 'none'; }
}

/* ==================== SPELL CHECK - TỰ ĐỘNG KIỂM TRA ==================== */
function checkSpellingForBackSide(){
    // Lấy từ đã lưu từ front template
    let userInput = localStorage.getItem('userSpellingInput');
    
    // Nếu không có trong localStorage, thử lấy từ các element khác
    if (!userInput) {
        const tg = document.querySelector('.typeGood');
        const tm = document.querySelector('.typeMissed');
        const tp = document.querySelector('.typePartial');
        if (tg) userInput = tg.textContent; 
        else if (tm) userInput = tm.textContent; 
        else if (tp) userInput = tp.textContent;
    } else {
        // Xóa sau khi đã lấy để tránh ảnh hưởng lần sau
        localStorage.removeItem('userSpellingInput');
    }

    const correctWord = document.querySelector('.word')?.textContent?.trim() || '';
    const resultDiv = document.getElementById('spellCheckResult');
    const hintDiv = document.getElementById('spellCheckHint');

    if (!userInput?.trim()){
        resultDiv.innerHTML = '<span style="color:#666;">Bạn chưa nhập từ nào ở mặt trước</span>';
        hintDiv.innerHTML = '💡 Hãy nhập từ ở mặt trước để kiểm tra chính tả';
        return;
    }

    const input = userInput.trim();
    const res = compareSpelling(correctWord, input);
    displaySpellingResult(res, correctWord, input);
}

// Hàm so sánh chính tả giữ nguyên
function compareSpelling(correct,input){
    const r = {
        isPerfectMatch: false,
        isPartialMatch: false,
        correctWord: correct,
        userInput: input,
        charResults: []
    };
    
    if (input.toLowerCase() === correct.toLowerCase()){
        r.isPerfectMatch = true;
        r.charResults = input.split('').map(c => ({char: c, status: 'correct', position: 'exact'}));
        return r;
    }
    
    const maxLength = Math.max(correct.length, input.length);
    for(let i = 0; i < maxLength; i++){
        if(i >= input.length) break;
        const ch = input[i];
        let status = 'incorrect';
        
        if(i < correct.length){
            const cc = correct[i];
            if(ch.toLowerCase() === cc.toLowerCase()) status = 'correct';
            else if(correct.toLowerCase().includes(ch.toLowerCase())) status = 'partial';
        }
        r.charResults.push({char: ch, status});
    }
    
    r.isPartialMatch = r.charResults.some(c => c.status !== 'incorrect');
    return r;
}

// Hàm hiển thị kết quả giữ nguyên
function displaySpellingResult(result, correctWord, userInput){
    const res = document.getElementById('spellCheckResult');
    const hint = document.getElementById('spellCheckHint');

    if(result.isPerfectMatch){
        res.innerHTML = userInput.split('').map(c => `<span class="spell-check-char correct-char">${c}</span>`).join('');
        hint.innerHTML = '✅ Chính xác! Từ được viết đúng chính tả.';
        return;
    }
    
    if(!result.isPartialMatch){
        res.innerHTML = userInput.split('').map(c => `<span class="spell-check-char incorrect-char">${c}</span>`).join('');
        hint.innerHTML = `❌ Sai hoàn toàn. Từ đúng là: <strong>${correctWord}</strong>`;
        return;
    }
    
    let html = '';
    result.charResults.forEach(cr => {
        let cls = 'spell-check-char ';
        if(cr.status === 'correct') cls += 'correct-char';
        else if(cr.status === 'partial') cls += 'partial-correct';
        else cls += 'incorrect-char';
        html += `<span class="${cls}">${cr.char}</span>`;
    });
    res.innerHTML = html;

    const correctCount = result.charResults.filter(c => c.status === 'correct').length;
    if(userInput.length < correctWord.length) {
        hint.innerHTML = `⚠️ Thiếu ký tự. Từ đúng có ${correctWord.length} ký tự.`;
    } else if(userInput.length > correctWord.length) {
        hint.innerHTML = `⚠️ Thừa ký tự. Từ đúng có ${correctWord.length} ký tự.`;
    } else {
        hint.innerHTML = `🔍 Có ${correctCount}/${correctWord.length} ký tự đúng. Từ đúng: <strong>${correctWord}</strong>`;
    }
}

    containerEl.innerHTML = `<div class="no-radicals-message"><div class="item-title">ℹ️ Thông tin</div><div class="item-content">Không có từ để phân tích.</div></div>`;
    return;
  }

  const raw = (dataEl?.textContent || '').trim();
  
  if (!raw || raw === '{}' || raw === '[]') {
    createDefaultRadicalAnalysis(mainWord, containerEl);
    return;
  }

  if (/^[\[{]/.test(raw)) {
    try {
      const data = JSON.parse(raw);
      renderRadicalsFromJSON(data, mainWord, containerEl);
    } catch (e) {
      console.warn('Radicals JSON parse failed, using default analysis:', e);
      createDefaultRadicalAnalysis(mainWord, containerEl);
    }
  } else {
    containerEl.innerHTML = `<div class="tab-item radicals-fallback">${raw}</div>`;
  }
}

function createDefaultRadicalAnalysis(word, containerEl) {
  let html = '';
  
  for (let i = 0; i < word.length; i++) {
    const char = word[i];
    
    html += `

function renderRadicalsFromJSON(data, mainWord, containerEl) {
  const arr = Array.isArray(data) ? data : [data];
  let finalHtml = '';

  // AI hiện trả về danh sách bộ thủ: {character, name_vi, name, meaning_vi, meaning, stroke_count, position, variant}
  // Dữ liệu kiểu cũ (phân tích từng chữ) có overall_explanation_vi/components -> vẫn hiển thị như trước
  const isRadicalItem = it => it && it.type !== 'memory_story' && !it.components && !it.overall_explanation_vi
    && (it.name_vi || it.meaning_vi || it.position || it.name || it.meaning);
  const radicalItems = arr.filter(isRadicalItem);

  if (radicalItems.length > 0) {
    finalHtml += `
      <div class="character-analysis">
        <div class="analysis-header">
          <h3 class="analysis-title">Bộ thủ trong chữ: <span class="analysis-char">${mainWord}</span></h3>
        </div>
        <div class="components-list">`;
    radicalItems.forEach(r => {
      const name = r.name_vi || r.name || '';
      const extraName = (r.name && r.name !== name) ? ` (${r.name})` : '';
      const strokes = parseInt(r.stroke_count, 10);
      finalHtml += `
          <div class="component-item">
            <div class="component-char">${r.character || ''}</div>
            <div class="component-details">
              <div class="component-header">
                <span class="component-pinyin">${name}${extraName}</span>
                ${strokes > 0 ? `<span class="component-role other">${strokes} nét</span>` : ''}
              </div>
              <div class="component-meaning">${r.meaning_vi || r.meaning || ''}</div>
              ${r.position ? `<div class="component-meaning">📍 Vị trí: ${r.position}</div>` : ''}
              ${r.variant ? `<div class="component-meaning">🔁 Biến thể: ${r.variant}</div>` : ''}
            </div>
          </div>`;
    });
    finalHtml += `
        </div>
      </div>`;
  }

  arr.forEach((analysis, index) => {
    if (analysis?.type === 'memory_story' || isRadicalItem(analysis)) return;

    const ch = analysis?.character?.trim() || mainWord[index] || mainWord;
    if (!ch) return;

    const strokeCount = (analysis?.stroke_count || getStrokeCount(ch)).toString().trim();
    
    finalHtml += `
      <div class="character-analysis">
        <div class="analysis-header">
          <h3 class="analysis-title">Phân tích chữ: <span class="analysis-char">${ch}</span></h3>
          ${strokeCount ? `
            <div class="char-stroke-count" onclick="openFanningStrokesModal('${ch}')">
              ✍️ ${strokeCount} nét
            </div>
          ` : ''}
        </div>
        
        <div class="character-construction">
          <div class="construction-title">🏗️ Quá trình hình thành chữ:</div>
          <div class="stroke-breakdown">
            ${analysis?.overall_explanation_vi || generateStrokeBreakdown(ch)}
          </div>
        </div>
    `;

    if (Array.isArray(analysis?.components) && analysis.components.length > 0 && !analysis.is_pictograph) {
      finalHtml += `<div class="components-list">`;
      analysis.components.forEach(comp => {
        if (!comp) return;
        const role = comp.role || '';
        const roleClass = role.includes('Nghĩa') ? 'semantic' : (role.includes('Âm') ? 'phonetic' : 'other');
        finalHtml += `
          <div class="component-item">
            <div class="component-char">${comp.component || ''}</div>
            <div class="component-details">
              <div class="component-header">
                <span class="component-pinyin">${comp.pinyin || ''}</span>
                <span class="component-role ${roleClass}">${role}</span>
              </div>
              <div class="component-meaning">${comp.meaning_vi || comp.meaning || ''}</div>
            </div>
          </div>
        `;
      });
      finalHtml += `</div>`;
    }

    finalHtml += `</div>`;
  });

  if (finalHtml) {
    containerEl.innerHTML = finalHtml;
  } else {
    // Chỉ có câu chuyện ghi nhớ (hoặc không có gì) -> phân tích mặc định, không gán kết quả undefined
    createDefaultRadicalAnalysis(mainWord, containerEl);
  }
}

// ==================== CẬP NHẬT HỆ THỐNG KHÓA CHO NÚT MỚI ====================
function initializeTabLock() {
    const LOCK_PASSWORD = "1";
    const STORAGE_KEY = "gem_section_unlocked_v1";
    const lockedSections = ['BoSung', 'MoRong', 'Radicals', 'ApDung'];

    // Kiểm tra trạng thái đã mở từ localStorage
    const savedUnlocked = localStorage.getItem(STORAGE_KEY) === "1";
    window.__gem_sections_unlocked = savedUnlocked;

    let currentTabToOpen = null;

    function applyLockState() {
        const unlocked = !!window.__gem_sections_unlocked;

        lockedSections.forEach(sectionId => {
            const element = document.getElementById(sectionId);
            if (!element) return;

            if (!element.classList.contains('gem-sec-wrap')) {
                element.classList.add('gem-sec-wrap');
            }

            if (unlocked) {
                element.classList.remove('gem-sec-locked');
                const existingOverlay = element.querySelector('.gem-sec-overlay');
                if (existingOverlay) {
                    existingOverlay.remove();
                }
                
                element.style.pointerEvents = 'auto';
                element.style.userSelect = 'auto';
                element.style.filter = 'none';
                element.style.webkitFilter = 'none';
            } else {
                element.classList.add('gem-sec-locked');
                if (!element.querySelector('.gem-sec-overlay')) {
                    const overlay = document.createElement('div');
                    overlay.className = 'gem-sec-overlay';
                    overlay.innerHTML = '🔒 ĐÃ KHOÁ<br><small>Nhấp vào tab và nhập mật khẩu để mở</small>';
                    element.appendChild(overlay);
                }
            }
        });

        updateTabLockIcons();
        updateLockButton();
    }

    function updateTabLockIcons() {
        const unlocked = !!window.__gem_sections_unlocked;
        const lockedTabNames = {
            'BoSung': '➕ Bổ sung',
            'MoRong': '🚀 Mở rộng', 
            'Radicals': '🖌️ Bộ',
            'ApDung': '📝 Áp dụng'
        };

        Object.keys(lockedTabNames).forEach(sectionId => {
            const tabLinks = document.querySelectorAll('.tab-link');
            tabLinks.forEach(tabLink => {
                const tabText = tabLink.textContent || tabLink.innerText;
                const targetText = lockedTabNames[sectionId];
                
                if (tabText.includes(targetText.replace(' 🔒', '').replace(' 🔓', ''))) {
                    // Biểu tượng 🔒/🔓 do CSS (.tab-link.locked::after) thêm vào, không chèn thêm vào chữ
                    tabLink.innerHTML = targetText;
                    tabLink.classList.toggle('locked', !unlocked);
                    tabLink.classList.toggle('unlocked', unlocked);
                }
            });
        });
    }

    function showPasswordModal(title, message, tabToOpen = null) {
        currentTabToOpen = tabToOpen;
        document.getElementById('gem-password-title').textContent = title;
        document.getElementById('gem-password-message').textContent = message;
        document.getElementById('gem-password-input').value = '';
        document.getElementById('auto-unlock-feedback').classList.remove('show');
        document.getElementById('password-error').classList.remove('show');
        document.getElementById('gem-password-modal').style.display = 'flex';
        
        setTimeout(() => {
            const input = document.getElementById('gem-password-input');
            input.focus();
        }, 100);
    }

    function hidePasswordModal() {
        document.getElementById('gem-password-modal').style.display = 'none';
        currentTabToOpen = null;
    }

    function showPasswordError() {
        const errorElement = document.getElementById('password-error');
        errorElement.classList.add('show');
        setTimeout(() => {
            errorElement.classList.remove('show');
        }, 3000);
    }

    function autoUnlock() {
        const feedback = document.getElementById('auto-unlock-feedback');
        feedback.classList.add('show');
        
        setTimeout(() => {
            window.__gem_sections_unlocked = true;
            localStorage.setItem(STORAGE_KEY, "1");
            applyLockState();
            hidePasswordModal();
            
            if (currentTabToOpen) {
                setTimeout(() => {
                    openTabDirectly(currentTabToOpen);
                }, 300);
            }
        }, 500);
    }

    function openTabDirectly(tabName) {
        const tabcontent = document.getElementsByClassName("tab-content");
        for (let i = 0; i < tabcontent.length; i++) {
            tabcontent[i].style.display = "none";
        }
        
        const tablinks = document.getElementsByClassName("tab-link");
        for (let i = 0; i < tablinks.length; i++) {
            tablinks[i].className = tablinks[i].className.replace(" active", "");
        }
        
        const targetTab = document.getElementById(tabName);
        if (targetTab) {
            targetTab.style.display = "block";
            
            const targetTabLink = document.querySelector(`.tab-link[onclick*="${tabName}"]`);
            if (targetTabLink) {
                targetTabLink.className += " active";
            }
        }
    }

    function setupPasswordModal() {
        const modal = document.getElementById('gem-password-modal');
        const input = document.getElementById('gem-password-input');
        const confirmBtn = document.getElementById('gem-password-confirm');
        const cancelBtn = document.getElementById('gem-password-cancel');

        input.addEventListener('input', function(e) {
            const password = e.target.value;
            if (password === LOCK_PASSWORD) {
                autoUnlock();
            }
        });

        const handleConfirm = function() {
            const password = input.value;
            if (password === LOCK_PASSWORD) {
                autoUnlock();
            } else {
                showPasswordError();
                input.value = '';
                input.focus();
            }
        };

        const handleCancel = function() {
            hidePasswordModal();
        };

        confirmBtn.addEventListener('click', handleConfirm);
        cancelBtn.addEventListener('click', handleCancel);

        input.addEventListener('keypress', function(e) {
            if (e.key === 'Enter') {
                handleConfirm();
            }
        });

        modal.addEventListener('click', function(e) {
            if (e.target === modal) {
                handleCancel();
            }
        });
    }

    function createLockButton() {
        if (document.getElementById('gem-lock-toggle-btn')) return;
        
        const lockBtn = document.createElement('button');
        lockBtn.id = 'gem-lock-toggle-btn';
        document.body.appendChild(lockBtn);
        
        const handleLockButtonClick = function() {
            if (window.__gem_sections_unlocked) {
                window.__gem_sections_unlocked = false;
                localStorage.setItem(STORAGE_KEY, "0");
                applyLockState();
            } else {
                showPasswordModal('🔐 MỞ KHOÁ', 'Nhập mật khẩu để mở khoá tất cả tab:');
            }
        };
        
        lockBtn.addEventListener('click', handleLockButtonClick);
        lockBtn.addEventListener('touchend', function(e) {
            e.preventDefault();
            handleLockButtonClick();
        });
        
        updateLockButton();
    }

    function updateLockButton() {
        const btn = document.getElementById('gem-lock-toggle-btn');
        if (!btn) return;
        
        if (window.__gem_sections_unlocked) {
            btn.innerHTML = '🔓 ĐANG MỞ<br>Bấm để khoá';
            btn.classList.remove('gem-locked');
            btn.classList.add('gem-open');
        } else {
            btn.innerHTML = '🔒 ĐANG KHOÁ<br>Bấm để mở';
            btn.classList.remove('gem-open');
            btn.classList.add('gem-locked');
        }
    }

    function setupTabInterception() {
        const tabLinks = document.querySelectorAll('.tab-link');
        tabLinks.forEach(link => {
            const onclickAttr = link.getAttribute('onclick') || '';
            const match = onclickAttr.match(/openTab\(event,\s*'(\w+)'\)/);
            if (!match) return;
            
            const targetTab = match[1];
            if (!lockedSections.includes(targetTab)) return;

            const handleTabClick = function(e) {
                if (!window.__gem_sections_unlocked) {
                    e.preventDefault();
                    e.stopPropagation();
                    showPasswordModal('🔐 MỞ KHOÁ', `Nhập mật khẩu để mở tab "${getTabDisplayName(targetTab)}":`, targetTab);
                }
            };

            link.addEventListener('click', handleTabClick);
            link.addEventListener('touchend', function(e) {
                if (!window.__gem_sections_unlocked) {
                    e.preventDefault();
                    handleTabClick(e);
                }
            });
        });
    }

    function getTabDisplayName(tabId) {
        const names = {
            'BoSung': 'Bổ sung',
            'MoRong': 'Mở rộng',
            'Radicals': 'Bộ',
            'LuyenCau': 'Luyện câu',
            'ApDung': 'Áp dụng'
        };
        return names[tabId] || tabId;
    }

    function initLockSystem() {
        createLockButton();
        setupPasswordModal();
        applyLockState();
        setupTabInterception();

        setTimeout(applyLockState, 500);
    }

    setTimeout(initLockSystem, 100);
}

    initializeTabLock();
}

// Hàm trích xuất Hán tự và Pinyin
function extractHanziAndPinyin(text) {
    let hanzi = text, pinyin = '';
    const matches = text.match(/\(([^)]+)\)/g);
    if (matches) {
        hanzi = text.replace(/\([^)]*\)/g, '').replace(/\s+/g, ' ').trim();
        pinyin = matches.map(m => m.slice(1, -1)).join(' ').replace(/\s+/g, ' ').trim();
    }
    return {hanzi, pinyin};
}

  const exampleEn=exEl.textContent;
  const mainWord=document.querySelector('.word')?.textContent||'';

  }
}

        if (res === 'started') return;
        btn.disabled = false;
        btn.textContent = '✨ Thêm chi tiết';
        note.textContent = res === 'word_mismatch' ? '⚠️ Thẻ đã đổi, mở lại thẻ rồi thử lại.' : '❌ Chưa thêm được chi tiết (' + res + ')';
    });
}
